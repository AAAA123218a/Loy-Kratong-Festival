local function setLocalPlayerFrozen(freeze)
    if freezeConnection then
        freezeConnection:Disconnect()
        freezeConnection = nil
    end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if not freeze then
        if savedPlayerState then
            if hum and hum.Parent then
                hum.WalkSpeed = savedPlayerState.WalkSpeed
                hum.JumpPower = savedPlayerState.JumpPower
                hum.JumpHeight = savedPlayerState.JumpHeight
                hum.AutoRotate = savedPlayerState.AutoRotate
            end
            if root and root.Parent then
                root.Anchored = savedPlayerState.Anchored
            end
            savedPlayerState = nil
        elseif root and root.Parent then
            root.Anchored = false
        end
        return
    end

    if not CONFIG.FreezeLocalPlayer then
        return
    end

    if hum and root then
        if not savedPlayerState then
            savedPlayerState = {
                WalkSpeed = hum.WalkSpeed,
                JumpPower = hum.JumpPower,
                JumpHeight = hum.JumpHeight,
                AutoRotate = hum.AutoRotate,
                Anchored = root.Anchored,
            }
        end
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum.JumpHeight = 0
        hum.AutoRotate = false
        root.Anchored = true
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end

    freezeConnection = RunService.Heartbeat:Connect(function()
        if not isEnabled or not CONFIG.FreezeLocalPlayer then
            return
        end
        local c = LocalPlayer.Character
        local h = c and c:FindFirstChildOfClass("Humanoid")
        local r = c and c:FindFirstChild("HumanoidRootPart")
        if h and h.Health > 0 then
            h.WalkSpeed = 0
            h.JumpPower = 0
            h.JumpHeight = 0
            h.AutoRotate = false
        end
        if r then
            r.Anchored = true
            r.AssemblyLinearVelocity = Vector3.zero
        end
    end)
end

-- ซ่อนตัวละครจริงของผู้เล่นอื่นเฉพาะมุมมองของเรา
-- =========================================================
local function setRealPlayersHidden(hidden)
    if hideConnection then
        hideConnection:Disconnect()
        hideConnection = nil
    end

    if not hidden then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                for _, desc in ipairs(player.Character:GetDescendants()) do
                    if desc:IsA("BasePart") then
                        desc.LocalTransparencyModifier = 0
                    end
                end
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hum and savedDisplayTypes[hum] then
                    hum.DisplayDistanceType = savedDisplayTypes[hum]
                end
            end
        end
        table.clear(savedDisplayTypes)
        return
    end

    if not CONFIG.HideRealPlayers then
        return
    end

    hideConnection = RunService.RenderStepped:Connect(function()
        if not isEnabled then
            return
        end
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                for _, desc in ipairs(player.Character:GetDescendants()) do
                    if desc:IsA("BasePart") then
                        desc.LocalTransparencyModifier = 1
                    end
                end
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hum then
                    if not savedDisplayTypes[hum] then
                        savedDisplayTypes[hum] = hum.DisplayDistanceType
                    end
                    hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                end
            end
        end
    end)
end

    base.Material = Enum.Material.SmoothPlastic
    base.CanCollide = false
    base.Anchored = true
    base.Massless = true
    base.CFrame = CFrame.new(0, 0, 0) * CFrame.Angles(0, 0, math.rad(90))
    base.Parent = model
    model.PrimaryPart = base

    local function addPart(name, size, color, material, worldCf, shape)
        local p = Instance.new("Part")
        p.Name = name
        p.Size = size
        p.Color = color
        p.Material = material or Enum.Material.SmoothPlastic
        p.CanCollide = false
        p.Anchored = true
        p.Massless = true
        if shape then
            p.Shape = shape
        end
        p.CFrame = worldCf
        p.Parent = model
        return p
    end

    -- ไส้หยวกกล้วยด้านบน
    addPart(
        "TrunkPith",
        Vector3.new(0.08, 1.25, 1.25),
        Color3.fromRGB(236, 239, 194),
        Enum.Material.SmoothPlastic,
        CFrame.new(0, 0.2, 0) * CFrame.Angles(0, 0, math.rad(90)),
        Enum.PartType.Cylinder
    )

    -- กลีบใบตองพับชั้นนอก 16 กลีบ
    local outerCount = 16
    for i = 1, outerCount do
        local angle = (i - 1) * ((math.pi * 2) / outerCount)
        local cf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.16, 0.8)
            * CFrame.Angles(math.rad(-32), 0, 0)
        addPart(
            "OuterLeaf_" .. i,
            Vector3.new(0.28, 0.56, 0.08),
            (i % 2 == 0) and Color3.fromRGB(56, 142, 60) or Color3.fromRGB(46, 125, 50),
            Enum.Material.SmoothPlastic,
            cf
        )
        -- ปลายกลีบใบตองแหลม
        local tipCf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.42, 0.88)
            * CFrame.Angles(math.rad(-18), 0, math.rad(45))
        addPart(
            "OuterTip_" .. i,
            Vector3.new(0.2, 0.2, 0.06),
            Color3.fromRGB(102, 187, 106),
            Enum.Material.SmoothPlastic,
            tipCf
        )
    end

    -- กลีบใบตองชั้นใน 10 กลีบ
    local innerCount = 10
    for i = 1, innerCount do
        local angle = (i - 1) * ((math.pi * 2) / innerCount) + 0.15
        local cf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.28, 0.56)
            * CFrame.Angles(math.rad(-22), 0, 0)
        addPart(
            "InnerLeaf_" .. i,
            Vector3.new(0.26, 0.4, 0.07),
            Color3.fromRGB(129, 199, 132),
            Enum.Material.SmoothPlastic,
            cf
        )
    end

    -- ดอกดาวเรืองรอบใน 6 ดอก
    for i = 1, 6 do
        local angle = (i - 1) * ((math.pi * 2) / 6)
        local cf = CFrame.Angles(0, angle, 0) * CFrame.new(0, 0.3, 0.42)
        addPart(
            "Marigold_" .. i,
            Vector3.new(0.24, 0.2, 0.24),
            (i % 2 == 0) and Color3.fromRGB(251, 191, 36) or Color3.fromRGB(245, 158, 11),
            Enum.Material.SmoothPlastic,

        local angle = (i - 1) * ((math.pi * 2) / 8)

-- =========================================================
-- ตัวช่วยจัดท่าทาง Motor6D สำหรับอนิเมชั่นลอยกระทงสมจริง (รองรับ R15 & R6)
-- =========================================================
local function snapshotCharacterMotors(char)
    local motors = {}
    for _, desc in ipairs(char:GetDescendants()) do
        if desc:IsA("Motor6D") then
            motors[desc.Name] = {
                motor = desc,
                origC0 = desc.C0,
            }
        end
    end
    return motors
end

local function restoreCharacterMotors(motors)
    for _, info in pairs(motors) do
        if info.motor and info.motor.Parent then
            info.motor.C0 = info.origC0
        end
    end
end

-- กำหนดมุมข้อต่อตามคีย์เฟรม: prayWeight (0..1), bendWeight (0..1)
local function applyCeremonyPose(motors, prayWeight, bendWeight)
    local armRaise = prayWeight * 0.72 + bendWeight * 0.58
    local armInward = prayWeight * 0.36 + bendWeight * 0.25
    local elbowBend = prayWeight * 0.65 - bendWeight * 0.2
    local headBow = prayWeight * 0.28 + bendWeight * 0.35
    local torsoBend = bendWeight * 0.55

    local function setAngle(name, cfAngle)
        local entry = motors[name]
        if entry and entry.motor and entry.motor.Parent then
            entry.motor.Transform = CFrame.identity
            entry.motor.C0 = entry.origC0 * cfAngle
        end
    end

    -- R15 joints
    setAngle("RightShoulder", CFrame.Angles(armRaise, 0, -armInward))
    setAngle("LeftShoulder", CFrame.Angles(armRaise, 0, armInward))
    setAngle("RightElbow", CFrame.Angles(math.max(0, elbowBend), 0, 0))
    setAngle("LeftElbow", CFrame.Angles(math.max(0, elbowBend), 0, 0))
    setAngle("Waist", CFrame.Angles(-torsoBend, 0, 0))
    setAngle("Neck", CFrame.Angles(-headBow, 0, 0))

    -- R6 fallback joints
    setAngle("Right Shoulder", CFrame.Angles(0, -armInward, armRaise))
    setAngle("Left Shoulder", CFrame.Angles(0, armInward, -armRaise))
    setAngle("RootJoint", CFrame.Angles(-torsoBend, 0, 0))
end

-- คำนวณตำแหน่งกระทงให้อยู่ระหว่างมือทั้งสองข้างตรงหน้าตัวละคร
local function positionKrathongInHands(krathongModel, char, prayWeight, bendWeight)
    if not krathongModel or not krathongModel.Parent or not char then
        return
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then
        return
    end

    local forwardDist = 1.65 + (bendWeight * 1.35)
    local heightOffset = 0.05 + (prayWeight * 0.55) - (bendWeight * 1.45)
    local yawOnly = root.CFrame - root.CFrame.Position
    local targetPos = (root.CFrame * CFrame.new(0, heightOffset, -forwardDist)).Position
    krathongModel:PivotTo(CFrame.new(targetPos) * yawOnly)
end

-- =========================================================
-- คำนวณความสูงพื้นจริงเพื่อไม่ให้ตัวละครจมพื้นหรือลอย
-- =========================================================
local function getFloorY(referenceRoot, x, z)
    local hum = referenceRoot.Parent and referenceRoot.Parent:FindFirstChildOfClass("Humanoid")
    local fallbackFloor = referenceRoot.Position.Y - (referenceRoot.Size.Y * 0.5) - (hum and hum.HipHeight or 2)

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local excludeList = {}
    if illusionFolder then
        table.insert(excludeList, illusionFolder)
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then
            table.insert(excludeList, p.Character)
        end
    end
    params.FilterDescendantsInstances = excludeList

    local origin = Vector3.new(x, referenceRoot.Position.Y + 6, z)
    local result = Workspace:Raycast(origin, Vector3.new(0, -40, 0), params)
    if result then
        return result.Position.Y
    end
    return fallbackFloor
end

-- =========================================================
-- สร้างโคลนตัวละครฝั่ง Client พร้อมรีเซ็ตเลือดเต็ม 100%
-- =========================================================
local function spawnFreshClone(sourcePlayer, cloneLabel, targetPos, facingYawCFrame, anchoredRoot)
    local sourceChar = sourcePlayer and sourcePlayer.Character
    if not sourceChar then
        return nil
    end

    local prevArch = sourceChar.Archivable
    sourceChar.Archivable = true
    local clone = sourceChar:Clone()
    sourceChar.Archivable = prevArch
    if not clone then
        return nil
    end

    clone.Name = cloneLabel or (sourcePlayer.Name .. "_Illusion")

    for _, desc in ipairs(clone:GetDescendants()) do
        if desc:IsA("Script") or desc:IsA("LocalScript") or desc:IsA("Tool") then
            desc:Destroy()
        elseif desc:IsA("BasePart") then
            desc.LocalTransparencyModifier = 0
            desc.CanCollide = false
            desc.Anchored = false
        end
    end

    local hum = clone:FindFirstChildOfClass("Humanoid")
    local root = clone:FindFirstChild("HumanoidRootPart") or clone.PrimaryPart
    if hum then
        hum.DisplayName = sourcePlayer.DisplayName
        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer
        hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOn
        hum.MaxHealth = 100
        hum.Health = 100
        hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
    end

    local folder = ensureIllusionFolder()
    clone.Parent = folder

    if root and targetPos then
        local hip = hum and hum.HipHeight or 2
        local floorY = targetPos.Y
        local exactRootY = floorY + hip + (root.Size.Y * 0.5)
        local finalCf = CFrame.new(targetPos.X, exactRootY, targetPos.Z) * (facingYawCFrame or CFrame.identity)
        clone:PivotTo(finalCf)
        if anchoredRoot then
            root.Anchored = true
        end
    end

    return clone
end

local function gatherTargetSources()
    local targets = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(targets, {
                player = player,
                label = player.DisplayName,
            })
        end
    end

    if #targets == 0 and CONFIG.SoloFallbackClones > 0 and LocalPlayer.Character then
        for i = 1, CONFIG.SoloFallbackClones do
            table.insert(targets, {
                player = LocalPlayer,
                label = LocalPlayer.DisplayName .. " #" .. i,
            })
        end
    end

    return targets
end

local function killCloneIllusion(clone)
    if not clone or not clone.Parent then
        return
    end
    local root = clone:FindFirstChild("HumanoidRootPart")
    if root then
        root.Anchored = false
    end
    local hum = clone:FindFirstChildOfClass("Humanoid")
    if hum then
        hum.Health = 0
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.Dead)
        end)
    end
    pcall(function()
        clone:BreakJoints()
    end)
    for _, part in ipairs(clone:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = true
            part.AssemblyLinearVelocity = Vector3.new(
                math.random(-8, 8),
                math.random(4, 11),
                math.random(-8, 8)
            )
        end
    end
end

    foam.CFrame = CFrame.new(shorePos.X, surfaceY + 0.1, shorePos.Z) * yawOnly
    foam.Parent = oceanModel

    -- แถบคลื่นน้ำขยับเบาๆ บนผิวน้ำ
    local waveParts = {}
    for w = 1, 5 do
        local distZ = -(10 + w * 26)
        local wPos = (anchorRoot.CFrame * CFrame.new(0, 0, distZ)).Position
        local wave = Instance.new("Part")
        wave.Name = "OceanWave_" .. w
        wave.Size = Vector3.new(CONFIG.OceanWidth * 0.92, 0.08, 3.2)
        wave.Color = Color3.fromRGB(186, 230, 253)
        wave.Material = Enum.Material.SmoothPlastic
        wave.Transparency = 0.7
        wave.Anchored = true
        wave.CanCollide = false
        wave.CFrame = CFrame.new(wPos.X, surfaceY + 0.16, wPos.Z) * yawOnly
        wave.Parent = oceanModel
        table.insert(waveParts, {
            part = wave,
            baseCf = wave.CFrame,
            phase = w * 1.1,
        })
    end

    local waveConn
    local tWave = 0
    waveConn = RunService.Heartbeat:Connect(function(dt)
        if not oceanModel.Parent then
            waveConn:Disconnect()
            return
        end
        tWave = tWave + dt
        for _, item in ipairs(waveParts) do
            local shift = math.sin(tWave * 1.6 + item.phase) * 1.8
            item.part.CFrame = item.baseCf * CFrame.new(0, math.sin(tWave * 2 + item.phase) * 0.04, shift)
        end
    end)

    return oceanModel, surfaceY
end

-- =========================================================
-- ปล่อยกระทงลงน้ำพร้อมระลอกคลื่นและลอยออกทะเลอย่างสมจริง
-- =========================================================
local function releaseKrathongToOcean(krathongModel, surfaceY, yawOnly, folder)
    if not krathongModel or not krathongModel.Parent then
        return
    end
    local primary = krathongModel.PrimaryPart
    if not primary then
        return
    end

    local startPos = Vector3.new(primary.Position.X, surfaceY + 0.22, primary.Position.Z)
    local startCf = CFrame.new(startPos) * yawOnly
    krathongModel:PivotTo(startCf)

    -- ระลอกคลื่นวงกลมตอนกระทงแตะผิวน้ำ
    local ripple = Instance.new("Part")
    ripple.Name = "WaterRipple"
    ripple.Shape = Enum.PartType.Cylinder
    ripple.Size = Vector3.new(0.05, 1.6, 1.6)
    ripple.Color = Color3.fromRGB(224, 242, 254)
    ripple.Material = Enum.Material.Neon
    ripple.Transparency = 0.45
    ripple.Anchored = true
    ripple.CanCollide = false
    ripple.CFrame = CFrame.new(startPos.X, surfaceY + 0.15, startPos.Z) * CFrame.Angles(0, 0, math.rad(90))
    ripple.Parent = folder

    TweenService:Create(
        ripple,
        TweenInfo.new(1.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {
            Size = Vector3.new(0.05, 5.2, 5.2),
            Transparency = 1,
        }
    ):Play()
    task.delay(1.15, function()
        if ripple and ripple.Parent then
            ripple:Destroy()
        end
    end)

    -- ขยับกระทงลอยคลื่นออกทะเลด้านหน้า
    task.spawn(function()
        local elapsed = 0
        local driftSpeed = 4.2 + (math.random() * 1.4)
        local bobSeed = math.random() * 6.28
        while krathongModel and krathongModel.Parent and isEnabled do
            local dt = RunService.Heartbeat:Wait()
            elapsed = elapsed + dt
            local forwardDist = elapsed * driftSpeed
            local bobY = math.sin(elapsed * 2.6 + bobSeed) * 0.08
            local swayX = math.sin(elapsed * 1.3 + bobSeed) * 0.35
            local targetCf = startCf * CFrame.new(swayX, bobY, -forwardDist)
            krathongModel:PivotTo(targetCf)
        end
    end)
end

-- เล่นอนิเมชั่นค่อยๆ ลอยกระทง (ใช้ได้ทั้งตัวเราและโคลนทุกคน)
local function playSlowKrathongAnimation(char, motors, krathongModel, surfaceY, folder, isSelf)
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        return
    end
    local yawOnly = root.CFrame - root.CFrame.Position

    -- ขั้นที่ 1 (0.0s -> 0.85s): ก้มศีรษะและยกกระทงขึ้นอธิษฐาน
    local t1 = 0
    while t1 < 0.85 and isEnabled do
        local dt = RunService.RenderStepped:Wait()
        t1 = math.min(0.85, t1 + dt)
        local alpha = math.sin((t1 / 0.85) * (math.pi * 0.5))
        applyCeremonyPose(motors, alpha, 0)
        positionKrathongInHands(krathongModel, char, alpha, 0)
    end

    if not isEnabled then
        return
    end

    -- ขั้นที่ 2 (0.85s -> 2.35s): ค่อยๆ โน้มตัวและลดแขนวางกระทงลงบนผิวน้ำ
    local t2 = 0
    while t2 < 1.5 and isEnabled do
        local dt = RunService.RenderStepped:Wait()
        t2 = math.min(1.5, t2 + dt)
        local alpha = t2 / 1.5
        local smooth = alpha * alpha * (3 - 2 * alpha)
        applyCeremonyPose(motors, 1 - smooth * 0.25, smooth)
        positionKrathongInHands(krathongModel, char, 1 - smooth * 0.25, smooth)
    end

    if not isEnabled then
        return
    end

    -- ปล่อยกระทงลงสู่ทะเลพร้อมระลอกคลื่น
    releaseKrathongToOcean(krathongModel, surfaceY, yawOnly, folder)
    if isSelf and activeSelfKrathongModel == krathongModel then
        activeSelfKrathongModel = nil
    end

    -- ขั้นที่ 3 (2.35s -> 3.15s): ค่อยๆ ยืดตัวกลับมายืนตรงมองกระทงลอยออกไป
    local t3 = 0
    while t3 < 0.8 and isEnabled do
        local dt = RunService.RenderStepped:Wait()
        t3 = math.min(0.8, t3 + dt)
        local remain = 1 - (t3 / 0.8)
        applyCeremonyPose(motors, remain * 0.5, remain)
    end
    restoreCharacterMotors(motors)
end

-- =========================================================
-- ล้างภาพลวงตาทั้งหมดและคืนค่าตัวละครเราทันทีเมื่อกดปิด
-- =========================================================
local function clearAllIllusions()
    if illusionFolder then
        illusionFolder:Destroy()
        illusionFolder = nil
    end
    if activeSelfKrathongModel then
        pcall(function()
            activeSelfKrathongModel:Destroy()
        end)
        activeSelfKrathongModel = nil
    end
    restoreCharacterMotors(savedSelfMotors)
    table.clear(savedSelfMotors)
    setRealPlayersHidden(false)
    setLocalPlayerFrozen(false)
end

-- =========================================================
-- ลูปหลัก 3 เฟส: กองตายด้านหลัง -> แถวหน้ากระดาน+ทะเล 3 วิ -> เราลอย+ทุกคนตาม 2 วิ
-- =========================================================
local function startIllusionLoop(currentToken)
    ensureIllusionFolder()
    setRealPlayersHidden(true)
    setLocalPlayerFrozen(true)

    task.spawn(function()
        while isEnabled and sessionToken == currentToken do
            local myChar = LocalPlayer.Character
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if not myRoot then
                task.wait(0.25)
                continue
            end

            setLocalPlayerFrozen(true)
            local anchorCf = myRoot.CFrame
            local yawOnly = anchorCf - anchorCf.Position

            -- ---------------------------------------------------------
            -- เฟส 1: ทุกคน (ยกเว้นเรา) กองรวมตายด้านหลังเรา / สุ่มตายทั่วแมพ
            -- ---------------------------------------------------------
            for burst = 1, CONFIG.DeathBurstCount do
                if not isEnabled or sessionToken ~= currentToken then
                    return
                end

                local folder = ensureIllusionFolder()
                folder:ClearAllChildren()

                local targets = gatherTargetSources()
                local burstClones = {}

                for _, entry in ipairs(targets) do
                    local offsetX, offsetZ
                    if CONFIG.DeathLocationMode == "random_map" then
                        offsetX = math.random(-55, 55)
                        offsetZ = math.random(-55, 55)
                    else
                        -- กองรวมกันด้านหลังเรา (+Z คือด้านหลังตัวละคร)
                        offsetX = math.random(-32, 32) / 10
                        offsetZ = math.random(65, 115) / 10
                    end

                    local worldXZ = (anchorCf * CFrame.new(offsetX, 0, offsetZ)).Position
                    local floorY = getFloorY(myRoot, worldXZ.X, worldXZ.Z)
                    local spawnPos = Vector3.new(worldXZ.X, floorY, worldXZ.Z)
                    local randomYaw = CFrame.Angles(0, math.rad(math.random(0, 359)), 0)

                    local clone = spawnFreshClone(entry.player, entry.label, spawnPos, randomYaw, false)
                    if clone then
                        table.insert(burstClones, clone)
                    end
                end

                task.wait(0.08)
                if not isEnabled or sessionToken ~= currentToken then
                    return
                end

                for _, clone in ipairs(burstClones) do
                    killCloneIllusion(clone)
                end

                task.wait(CONFIG.DeathIntervalSec)
            end

            if not isEnabled or sessionToken ~= currentToken then
                return
            end

            -- ---------------------------------------------------------
            -- เฟส 2: เกิดใหม่เลือดเต็ม 100% วาปเรียงแถวหน้ากระดาน + เสกทะเล 3 วิ
            -- ---------------------------------------------------------
            local folder = ensureIllusionFolder()
            folder:ClearAllChildren()

            local targets = gatherTargetSources()
            local lineupEntries = {}

            for idx, entry in ipairs(targets) do
                -- แถวหน้ากระดานแถวเดียวกับเรา (สลับซ้าย-ขวาแนวนอน)
                local tier = math.ceil(idx / 2)
                local side = (idx % 2 == 1) and -1 or 1
                local xOffset = side * tier * CONFIG.LineupSpacing
                local slotWorld = (anchorCf * CFrame.new(xOffset, 0, 0)).Position
                local slotFloorY = getFloorY(myRoot, slotWorld.X, slotWorld.Z)
                local spawnPos = Vector3.new(slotWorld.X, slotFloorY, slotWorld.Z)

                local clone = spawnFreshClone(entry.player, entry.label, spawnPos, yawOnly, true)
                if clone then
                    local motors = snapshotCharacterMotors(clone)
                    applyCeremonyPose(motors, 0.18, 0)

                    local krathong = createRealisticKrathongModel("Krathong_" .. idx)
                    krathong.Parent = folder
                    positionKrathongInHands(krathong, clone, 0.18, 0)

                    table.insert(lineupEntries, {
                        clone = clone,
                        motors = motors,
                        krathong = krathong,
                    })
                end
            end

            -- ตัวเราถือกระทงสมจริงด้วย
            savedSelfMotors = snapshotCharacterMotors(myChar)
            if CONFIG.GiveSelfKrathong then
                applyCeremonyPose(savedSelfMotors, 0.18, 0)
                activeSelfKrathongModel = createRealisticKrathongModel("Krathong_Self")
                activeSelfKrathongModel.Parent = folder
                positionKrathongInHands(activeSelfKrathongModel, myChar, 0.18, 0)
            end

            -- เสกทะเลขนาดใหญ่หน้าเรา รอ 3 วินาที
            local _, oceanSurfaceY = spawnHugeOceanInFront(myRoot, folder)
            local waitElapsed = 0
            while waitElapsed < CONFIG.OceanWaitSec and isEnabled and sessionToken == currentToken do
                local dt = RunService.RenderStepped:Wait()
                waitElapsed = waitElapsed + dt
                if activeSelfKrathongModel then
                    applyCeremonyPose(savedSelfMotors, 0.18, 0)
                    positionKrathongInHands(activeSelfKrathongModel, myChar, 0.18, 0)
                end
            end

            if not isEnabled or sessionToken ~= currentToken then
                return
            end

            -- ---------------------------------------------------------
            -- เฟส 3: เราเริ่มลอยกระทงสมจริง -> อีก 2 วิทุกคนทำตามพร้อมกันหมด
            -- ---------------------------------------------------------
            task.spawn(function()
                if activeSelfKrathongModel then
                    playSlowKrathongAnimation(
                        myChar,
                        savedSelfMotors,
                        activeSelfKrathongModel,
                        oceanSurfaceY,
                        folder,
                        true
                    )
                end
            end)

            local delayElapsed = 0
            while delayElapsed < CONFIG.GroupFollowDelaySec and isEnabled and sessionToken == currentToken do
                local dt = RunService.RenderStepped:Wait()
                delayElapsed = delayElapsed + dt
                for _, item in ipairs(lineupEntries) do
                    applyCeremonyPose(item.motors, 0.18, 0)
                    positionKrathongInHands(item.krathong, item.clone, 0.18, 0)
                end
            end

            if not isEnabled or sessionToken ~= currentToken then
                return
            end

            -- ทุกคนในแถวหน้ากระดานทำตามพร้อมกันหมด
            for _, item in ipairs(lineupEntries) do
                task.spawn(function()
                    playSlowKrathongAnimation(
                        item.clone,
                        item.motors,
                        item.krathong,
                        oceanSurfaceY,
                        folder,
                        false
                    )
                end)
            end

            -- รอชมกระทงลอยออกทะเลก่อนวนกลับรอบใหม่
            local watchElapsed = 0
            local totalWatch = 3.15 + CONFIG.FloatWatchSec
            while watchElapsed < totalWatch and isEnabled and sessionToken == currentToken do
                local dt = RunService.Heartbeat:Wait()
                watchElapsed = watchElapsed + dt
            end

            if not CONFIG.LoopCycle then
                break
            end
        end
    end)
end

        toggleBtn.BackgroundColor3 = Color3.fromRGB(180, 83, 9)
        stroke.Color = Color3.fromRGB(251, 191, 36)
        toggleBtn.Text = "กระทง: เปิด"
    else
        toggleBtn.BackgroundColor3 = Color3.fromRGB(17, 24, 39)
        stroke.Color = Color3.fromRGB(71, 85, 105)
        toggleBtn.Text = "กระทง: ปิด"
    end
end

local dragging = false
local dragStart = nil
local startPos = nil
local movedDistance = 0

toggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = toggleBtn.Position
        movedDistance = 0
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        movedDistance = delta.Magnitude
        toggleBtn.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if dragging then
            dragging = false
            if movedDistance < 6 then
                isEnabled = not isEnabled
                sessionToken = sessionToken + 1
                updateButtonVisual()
                if isEnabled then
                    startIllusionLoop(sessionToken)
                else
                    clearAllIllusions()
                end
            end
        end
    end
end)

updateButtonVisual()
