-- ระบบโหลดเพลงเตรียมไว้ล่วงหน้า & เล่นวนซ้ำทันทีตอนกดเปิด
-- =========================================================
local function resolveAudioAsset()
    if cachedAudioAsset then
        return cachedAudioAsset
    end

    local getAsset = (type(getcustomasset) == "function" and getcustomasset)
        or (type(getsynasset) == "function" and getsynasset)

    if not getAsset or type(writefile) ~= "function" then
        return nil
    end

    local okGet, audioBytes = pcall(function()
        return game:HttpGet(CONFIG.MusicUrl)
    end)
    if not okGet or type(audioBytes) ~= "string" or #audioBytes < 128 then
        return nil
    end

    local okWrite = pcall(function()
        writefile(MUSIC_FILE, audioBytes)
    end)
    if not okWrite then
        return nil
    end

    local okAsset, assetUri = pcall(getAsset, MUSIC_FILE)
    if okAsset and type(assetUri) == "string" and assetUri ~= "" then
        cachedAudioAsset = assetUri
        return cachedAudioAsset
    end

    return nil
end

local function stopBackgroundMusic()
    if activeSound then
        pcall(function()
            activeSound:Stop()
            activeSound:Destroy()
        end)
        activeSound = nil
    end
end

local function startBackgroundMusic(currentToken)
    stopBackgroundMusic()
    if not CONFIG.EnableMusic then
        return nil
    end

    local sound = Instance.new("Sound")
    sound.Name = "LoyKratongFestival_BGM"
    sound.Looped = true
    sound.Volume = CONFIG.MusicVolume
    sound.Parent = SoundService
    activeSound = sound

    task.spawn(function()
        local assetId = resolveAudioAsset()
        if not assetId or not isEnabled or sessionToken ~= currentToken or activeSound ~= sound then
            return
        end
        sound.SoundId = assetId
        pcall(function()
            sound:Play()
        end)
    end)

    return sound
end

local function getSyncedSongDuration()
    if activeSound and activeSound.Parent and activeSound.TimeLength and activeSound.TimeLength > 2 then
        return activeSound.TimeLength
    end
    return CONFIG.FallbackCycleSec
end

local function setCameraStage(phaseName, anchorCf, lineupCount)
    if currentCamPhase ~= phaseName then
        camPhaseClock = 0
    end
    currentCamPhase = phaseName
    if anchorCf then
        currentCamAnchor = anchorCf
    end
    if lineupCount then
        currentLineupCount = lineupCount
    end
end

local function setCinematicCameraEnabled(enabled)
    if cameraConnection then
        cameraConnection:Disconnect()
        cameraConnection = nil
    end

    local cam = Workspace.CurrentCamera
    if not cam then
        return
    end

    if not enabled then
        if savedCameraState then
            cam.CameraType = savedCameraState.CameraType
            cam.FieldOfView = savedCameraState.FieldOfView
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            cam.CameraSubject = hum or savedCameraState.CameraSubject
            savedCameraState = nil
        else
            cam.CameraType = Enum.CameraType.Custom
        end
        return
    end

    if not CONFIG.EnableCinematicCamera then
        return
    end

    if not savedCameraState then
        savedCameraState = {
            CameraType = cam.CameraType,
            CameraSubject = cam.CameraSubject,
            FieldOfView = cam.FieldOfView,
        }
    end

    cam.CameraType = Enum.CameraType.Scriptable

    cameraConnection = RunService.RenderStepped:Connect(function(dt)
        if not isEnabled or not CONFIG.EnableCinematicCamera then
            return
        end
        local activeCam = Workspace.CurrentCamera
        if not activeCam then
            return
        end
        if activeCam.CameraType ~= Enum.CameraType.Scriptable then
            activeCam.CameraType = Enum.CameraType.Scriptable
        end

        camPhaseClock = camPhaseClock + dt
        local baseCf = currentCamAnchor
        local halfSpan = math.max(1, math.ceil(currentLineupCount / 2)) * CONFIG.LineupSpacing
        local targetCf = activeCam.CFrame
        local targetFov = 60

        if currentCamPhase == "act1_ramwong_krathong" then
            local panX = math.sin(camPhaseClock * 0.45) * math.min(9, halfSpan * 0.6)
            local distZ = math.max(18, 12 + halfSpan * 0.85)
            local camPos = (baseCf * CFrame.new(panX, 5.8, -distZ)).Position
            local lookAt = (baseCf * CFrame.new(0, 2.2, 0)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 58
        elseif currentCamPhase == "act2_self_release" then
            local camPos = (baseCf * CFrame.new(-3.4, 3.0, -7.2)).Position
            local lookAt = (baseCf * CFrame.new(0, 1.3, -1.4)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 50
        elseif currentCamPhase == "act2_group_release" then
            local distZ = math.max(22, 16 + halfSpan * 0.9)
            local camPos = (baseCf * CFrame.new(math.sin(camPhaseClock * 0.4) * 3.5, 8.2, -distZ)).Position
            local lookAt = (baseCf * CFrame.new(0, 1.4, -5.0)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 63
        elseif currentCamPhase == "act3_celebration_dance" then
            local orbitX = math.sin(camPhaseClock * 0.38) * math.min(13, halfSpan * 0.85 + 4)
            local distZ = math.max(20, 14 + halfSpan * 0.85)
            local heightY = 6.2 + math.sin(camPhaseClock * 0.5) * 1.4
            local camPos = (baseCf * CFrame.new(orbitX, heightY, -distZ)).Position
            local lookAt = (baseCf * CFrame.new(0, 2.2, -2.5)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 64
        elseif currentCamPhase == "act4_finale_wave" then
            local distZ = math.max(26, 18 + halfSpan * 0.95)
            local camPos = (baseCf * CFrame.new(0, 10.5, -distZ)).Position
            local lookAt = (baseCf * CFrame.new(0, 2.6, -8.0)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 66
        end

        local alpha = math.clamp(dt * 3.2, 0, 1)
        activeCam.CFrame = activeCam.CFrame:Lerp(targetCf, alpha)
        activeCam.FieldOfView = activeCam.FieldOfView + (targetFov - activeCam.FieldOfView) * alpha
    end)
end

-- ระบบล็อกตัวเราให้อยู่กลางแถวหน้ากระดาน (คืนค่าปกติเมื่อกดปิด)

local function setLocalPlayerFrozen(freeze)
    if freezeConnection then
        freezeConnection:Disconnect()
        freezeConnection = nil
    end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if not freeze then
        if savedPlayerState then
            if hum and hum.Parent then
                hum.WalkSpeed = savedPlayerState.WalkSpeed
                hum.JumpPower = savedPlayerState.JumpPower
                hum.JumpHeight = savedPlayerState.JumpHeight
                hum.AutoRotate = savedPlayerState.AutoRotate
            end
            if root and root.Parent then
                root.Anchored = savedPlayerState.Anchored
            end
            savedPlayerState = nil
        elseif root and root.Parent then
            root.Anchored = false
        end
        return
    end

    if not CONFIG.FreezeLocalPlayer then
        return
    end

    if hum and root then
        if not savedPlayerState then
            savedPlayerState = {
                WalkSpeed = hum.WalkSpeed,
                JumpPower = hum.JumpPower,
                JumpHeight = hum.JumpHeight,
                AutoRotate = hum.AutoRotate,
                Anchored = root.Anchored,
            }
        end
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum.JumpHeight = 0
        hum.AutoRotate = false
        root.Anchored = true
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end

    freezeConnection = RunService.Heartbeat:Connect(function()
        if not isEnabled or not CONFIG.FreezeLocalPlayer then
            return
        end
        local c = LocalPlayer.Character
        local h = c and c:FindFirstChildOfClass("Humanoid")
        local r = c and c:FindFirstChild("HumanoidRootPart")
        if h and h.Health > 0 then
            h.WalkSpeed = 0
            h.JumpPower = 0
            h.JumpHeight = 0
            h.AutoRotate = false
        end
        if r then
            r.Anchored = true
            r.AssemblyLinearVelocity = Vector3.zero
        end
    end)
end

-- =========================================================
-- ซ่อนตัวละครจริงของผู้เล่นอื่นเฉพาะมุมมองของเรา
-- =========================================================
local function setRealPlayersHidden(hidden)
    if hideConnection then
        hideConnection:Disconnect()
        hideConnection = nil
    end

    if not hidden then
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                for _, desc in ipairs(player.Character:GetDescendants()) do
                    if desc:IsA("BasePart") then
                        desc.LocalTransparencyModifier = 0
                    end
                end
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hum and savedDisplayTypes[hum] then
                    hum.DisplayDistanceType = savedDisplayTypes[hum]
                end
            end
        end
        table.clear(savedDisplayTypes)
        return
    end

    if not CONFIG.HideRealPlayers then
        return
    end

    hideConnection = RunService.RenderStepped:Connect(function()
        if not isEnabled then
            return
        end
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                for _, desc in ipairs(player.Character:GetDescendants()) do
                    if desc:IsA("BasePart") then
                        desc.LocalTransparencyModifier = 1
                    end
                end
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hum then
                    if not savedDisplayTypes[hum] then
                        savedDisplayTypes[hum] = hum.DisplayDistanceType
                    end
                    hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                end
            end
        end
    end)
end

local function createRealisticKrathongModel(modelName)
    local model = Instance.new("Model")
    model.Name = modelName or "RealisticKrathong"

    local pivot = Instance.new("Part")
    pivot.Name = "PivotPart"
    pivot.Size = Vector3.new(0.2, 0.2, 0.2)
    pivot.Transparency = 1
    pivot.CanCollide = false
    pivot.Anchored = true
    pivot.Massless = true
    pivot.CFrame = CFrame.new(0, 0, 0)
    pivot.Parent = model
    model.PrimaryPart = pivot

    local function addPart(name, size, color, material, worldCf, shape)
        local p = Instance.new("Part")
        p.Name = name
        p.Size = size
        p.Color = color
        p.Material = material or Enum.Material.SmoothPlastic
        p.CanCollide = false
        p.Anchored = true
        p.Massless = true
        if shape then
            p.Shape = shape
        end
        p.CFrame = worldCf
        p.Parent = model
        return p
    end

    addPart(
        "Base",
        Vector3.new(0.42, 1.55, 1.55),
        Color3.fromRGB(34, 84, 38),
        Enum.Material.SmoothPlastic,
        CFrame.new(0, 0, 0) * CFrame.Angles(0, 0, math.rad(90)),
        Enum.PartType.Cylinder
    )

    addPart(
        "TrunkPith",
        Vector3.new(0.08, 1.25, 1.25),
        Color3.fromRGB(236, 239, 194),
        Enum.Material.SmoothPlastic,
        CFrame.new(0, 0.2, 0) * CFrame.Angles(0, 0, math.rad(90)),
        Enum.PartType.Cylinder
    )

    local outerCount = 16
    for i = 1, outerCount do
        local angle = (i - 1) * ((math.pi * 2) / outerCount)
        local cf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.16, 0.8)
            * CFrame.Angles(math.rad(-32), 0, 0)
        addPart(
            "OuterLeaf_" .. i,
            Vector3.new(0.28, 0.56, 0.08),
            (i % 2 == 0) and Color3.fromRGB(56, 142, 60) or Color3.fromRGB(46, 125, 50),
            Enum.Material.SmoothPlastic,
            cf
        )
        local tipCf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.42, 0.88)
            * CFrame.Angles(math.rad(-18), 0, math.rad(45))
        addPart(
            "OuterTip_" .. i,
            Vector3.new(0.2, 0.2, 0.06),
            Color3.fromRGB(102, 187, 106),
            Enum.Material.SmoothPlastic,
            tipCf
        )
    end

    local innerCount = 10
    for i = 1, innerCount do
        local angle = (i - 1) * ((math.pi * 2) / innerCount) + 0.15
        local cf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.28, 0.56)
            * CFrame.Angles(math.rad(-22), 0, 0)
        addPart(
            "InnerLeaf_" .. i,
            Vector3.new(0.26, 0.4, 0.07),
            Color3.fromRGB(129, 199, 132),
            Enum.Material.SmoothPlastic,
            cf
        )
    end

    for i = 1, 6 do
        local angle = (i - 1) * ((math.pi * 2) / 6)
        local cf = CFrame.Angles(0, angle, 0) * CFrame.new(0, 0.3, 0.42)
        addPart(
            "Marigold_" .. i,
            Vector3.new(0.24, 0.2, 0.24),
            (i % 2 == 0) and Color3.fromRGB(251, 191, 36) or Color3.fromRGB(245, 158, 11),
            Enum.Material.SmoothPlastic,
            cf,
            Enum.PartType.Ball
        )
    end

    for i = 1, 8 do
        local angle = (i - 1) * ((math.pi * 2) / 8)
        local cf = CFrame.Angles(0, angle, 0)
            * CFrame.new(0, 0.33, 0.22)
            * CFrame.Angles(math.rad(-28), 0, 0)
        addPart(
            "LotusPetal_" .. i,
            Vector3.new(0.18, 0.3, 0.07),
            (i % 2 == 0) and Color3.fromRGB(244, 114, 182) or Color3.fromRGB(251, 207, 232),
            Enum.Material.SmoothPlastic,
            cf
        )
    end

    addPart(
        "CandleBody",
        Vector3.new(0.13, 0.58, 0.13),
        Color3.fromRGB(253, 224, 71),
        Enum.Material.SmoothPlastic,
        CFrame.new(0, 0.52, 0)
    )

    local flame = addPart(
        "CandleFlame",
        Vector3.new(0.11, 0.2, 0.11),
        Color3.fromRGB(255, 183, 77),
        Enum.Material.Neon,
        CFrame.new(0, 0.88, 0),
        Enum.PartType.Ball
    )

    local light = Instance.new("PointLight")
    light.Color = Color3.fromRGB(255, 195, 90)
    light.Range = 11
    light.Brightness = 2.0
    light.Parent = flame

    local fire = Instance.new("Fire")
    fire.Size = 1.2
    fire.Heat = 2.0
    fire.Color = Color3.fromRGB(255, 200, 80)
    fire.SecondaryColor = Color3.fromRGB(255, 100, 30)

    for i = -1, 1 do
        local incenseCf = CFrame.new(i * 0.09, 0.52, -0.13)
            * CFrame.Angles(math.rad(-10), 0, math.rad(i * 6))
        addPart(
            "IncenseStick_" .. (i + 2),

            Color3.fromRGB(161, 98, 7),

            "IncenseEmber_" .. (i + 2),
            Vector3.new(0.045, 0.05, 0.045),
            Color3.fromRGB(255, 90, 30),
            Enum.Material.Neon,
            incenseCf * CFrame.new(0, 0.34, 0),
            Enum.PartType.Ball
        )
        if i == 0 then
            local smoke = Instance.new("ParticleEmitter")
            smoke.Rate = 6
            smoke.Lifetime = NumberRange.new(1.0, 1.8)
            smoke.Speed = NumberRange.new(0.5, 1.0)

                NumberSequenceKeypoint.new(1, 0.35),
            })
            smoke.Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.55),
                NumberSequenceKeypoint.new(1, 1),
            })
            smoke.Color = ColorSequence.new(Color3.fromRGB(226, 232, 240))
            smoke.Parent = ember
        end
    end

    return model
end

local function spawnSkyLantern(anchorRoot, folder)
    if not CONFIG.EnableSkyLanterns then
        return
    end
    local yawOnly = anchorRoot.CFrame - anchorRoot.CFrame.Position
    local offsetX = math.random(-65, 65)
    local offsetZ = -math.random(25, 110)
    local startPos = (anchorRoot.CFrame * CFrame.new(offsetX, math.random(2, 6), offsetZ)).Position

    local lantern = Instance.new("Part")
    lantern.Name = "SkyLantern"
    lantern.Size = Vector3.new(1.6, 2.2, 1.6)
    lantern.Color = Color3.fromRGB(255, 185, 75)
    lantern.Material = Enum.Material.Neon
    lantern.Transparency = 0.15
    lantern.Anchored = true
    lantern.CanCollide = false
    lantern.CFrame = CFrame.new(startPos) * yawOnly
    lantern.Parent = folder

    local light = Instance.new("PointLight")
    light.Color = Color3.fromRGB(255, 190, 85)
    light.Range = 14
    light.Brightness = 1.6
    light.Parent = lantern

    task.spawn(function()
        local t = 0
        local speedY = 2.4 + math.random() * 1.6
        local swaySeed = math.random() * 6.28
        while lantern and lantern.Parent and isEnabled and t < 16 do
            local dt = RunService.Heartbeat:Wait()
            t = t + dt
            local curPos = startPos + Vector3.new(
                math.sin(t * 0.9 + swaySeed) * 2.2,
                t * speedY,
                math.cos(t * 0.7 + swaySeed) * 1.8
            )
            lantern.CFrame = CFrame.new(curPos) * yawOnly * CFrame.Angles(0, t * 0.35, 0)
            if t > 13 then
                lantern.Transparency = math.min(1, 0.15 + ((t - 13) / 3) * 0.85)
            end
        end
        if lantern and lantern.Parent then
            lantern:Destroy()
        end
    end)
end

-- =========================================================
-- ระบบข้อต่อ Motor6D: ท่ารำวงถือกระทง / ท่าลอยกระทง / ท่าเต้นฉลองสนุกๆ
-- =========================================================
local function snapshotCharacterMotors(char)
    local motors = {}
    for _, desc in ipairs(char:GetDescendants()) do
        if desc:IsA("Motor6D") then
            motors[desc.Name] = {
                motor = desc,
                origC0 = desc.C0,
            }
        end
    end
    return motors
end

local function restoreCharacterMotors(motors)
    for _, info in pairs(motors) do
        if info.motor and info.motor.Parent then
            info.motor.C0 = info.origC0
        end
    end
end

local function setMotorAngle(motors, name, cfAngle)
    local entry = motors[name]
    if entry and entry.motor and entry.motor.Parent then
        entry.motor.Transform = CFrame.identity
        entry.motor.C0 = entry.origC0 * cfAngle
    end
end

-- ช่วงที่ 1: รำวงโยกตัวเข้าจังหวะเพลงขณะถือกระทงไว้ข้างหน้า
local function applyRamwongHoldPose(motors, timeSec, phaseOffset)
    local beat = (timeSec * 3.1 * CONFIG.DanceEnergy) + (phaseOffset or 0)
    local sway = math.sin(beat)
    local bounce = math.abs(math.cos(beat))

    local armBase = 0.65 + bounce * 0.14
    local armInward = 0.36
    local waistYaw = sway * 0.18
    local waistRoll = sway * 0.09
    local headTilt = sway * 0.12

    setMotorAngle(motors, "RightShoulder", CFrame.Angles(armBase, 0, -armInward + sway * 0.06))
    setMotorAngle(motors, "LeftShoulder", CFrame.Angles(armBase, 0, armInward + sway * 0.06))
    setMotorAngle(motors, "RightElbow", CFrame.Angles(0.62, 0, 0))
    setMotorAngle(motors, "LeftElbow", CFrame.Angles(0.62, 0, 0))
    setMotorAngle(motors, "Waist", CFrame.Angles(-0.04, waistYaw, waistRoll))
    setMotorAngle(motors, "Neck", CFrame.Angles(-0.05 + bounce * 0.06, -waistYaw * 0.4, headTilt))

    -- R6
    setMotorAngle(motors, "Right Shoulder", CFrame.Angles(0, -armInward, armBase))
    setMotorAngle(motors, "Left Shoulder", CFrame.Angles(0, armInward, -armBase))
    setMotorAngle(motors, "RootJoint", CFrame.Angles(-0.04, waistRoll, waistYaw))
end

-- ช่วงที่ 2: ท่าอธิษฐานและค่อยๆ ก้มปล่อยกระทงลงทะเล
local function applyCeremonyPose(motors, prayWeight, bendWeight)
    local armRaise = prayWeight * 0.72 + bendWeight * 0.58
    local armInward = prayWeight * 0.36 + bendWeight * 0.25
    local elbowBend = prayWeight * 0.65 - bendWeight * 0.2
    local headBow = prayWeight * 0.28 + bendWeight * 0.35
    local torsoBend = bendWeight * 0.55

    setMotorAngle(motors, "RightShoulder", CFrame.Angles(armRaise, 0, -armInward))
    setMotorAngle(motors, "LeftShoulder", CFrame.Angles(armRaise, 0, armInward))
    setMotorAngle(motors, "RightElbow", CFrame.Angles(math.max(0, elbowBend), 0, 0))
    setMotorAngle(motors, "LeftElbow", CFrame.Angles(math.max(0, elbowBend), 0, 0))
    setMotorAngle(motors, "Waist", CFrame.Angles(-torsoBend, 0, 0))
    setMotorAngle(motors, "Neck", CFrame.Angles(-headBow, 0, 0))

    -- R6
    setMotorAngle(motors, "Right Shoulder", CFrame.Angles(0, -armInward, armRaise))
    setMotorAngle(motors, "Left Shoulder", CFrame.Angles(0, armInward, -armRaise))
    setMotorAngle(motors, "RootJoint", CFrame.Angles(-torsoBend, 0, 0))
end

-- ช่วงที่ 3: เต้นรำวงวันลอยกระทงและโบกมือฉลองสนุกๆ ริมทะเล
local function applyFestivalDancePose(motors, timeSec, phaseOffset)
    local beat = (timeSec * 3.8 * CONFIG.DanceEnergy) + (phaseOffset or 0)
    local sway = math.sin(beat)
    local altSway = math.cos(beat)
    local pulse = math.sin(beat * 2)

    -- รำวงจีบมือสลับขึ้นลงซ้าย-ขวาอย่างสนุกสนาน
    local rightRaise = 1.45 + sway * 0.55
    local leftRaise = 1.45 - sway * 0.55
    local rightOut = 0.45 + altSway * 0.25
    local leftOut = -0.45 + altSway * 0.25
    local elbowR = 0.65 + pulse * 0.25
    local elbowL = 0.65 - pulse * 0.25
    local waistYaw = sway * 0.32
    local waistRoll = altSway * 0.14
    local headBob = pulse * 0.12

    setMotorAngle(motors, "RightShoulder", CFrame.Angles(rightRaise, 0, rightOut))
    setMotorAngle(motors, "LeftShoulder", CFrame.Angles(leftRaise, 0, leftOut))
    setMotorAngle(motors, "RightElbow", CFrame.Angles(math.max(0.15, elbowR), 0, 0))
    setMotorAngle(motors, "LeftElbow", CFrame.Angles(math.max(0.15, elbowL), 0, 0))
    setMotorAngle(motors, "Waist", CFrame.Angles(-0.06 + pulse * 0.06, waistYaw, waistRoll))
    setMotorAngle(motors, "Neck", CFrame.Angles(headBob, -waistYaw * 0.45, sway * 0.14))

    -- R6
    setMotorAngle(motors, "Right Shoulder", CFrame.Angles(0, rightOut * 0.4, rightRaise))
    setMotorAngle(motors, "Left Shoulder", CFrame.Angles(0, leftOut * 0.4, -leftRaise))
    setMotorAngle(motors, "RootJoint", CFrame.Angles(-0.05 + pulse * 0.05, waistRoll, waistYaw))
end

local function positionKrathongInHands(krathongModel, char, prayWeight, bendWeight, swayOffset)
    if not krathongModel or not krathongModel.Parent or not char then
        return
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    local chest = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or root
    if not root or not chest then
        return
    end

    local forwardDist = 1.55 + (bendWeight * 1.35)
    local heightOffset = 0.08 + (prayWeight * 0.45) - (bendWeight * 1.45)
    local yawOnly = root.CFrame - root.CFrame.Position
    local anchorRef = CFrame.new(chest.Position) * yawOnly
    local targetPos = (anchorRef * CFrame.new(swayOffset or 0, heightOffset, -forwardDist)).Position
    krathongModel:PivotTo(CFrame.new(targetPos) * yawOnly)
end

-- =========================================================
-- ความสูงพื้นรองรับทั้ง R6 และ R15 (ไม่จมพื้น)
-- =========================================================
local function getRootOffsetFromFloor(char, hum, root)
    if hum and hum.RigType == Enum.HumanoidRigType.R6 then
        local leg = char:FindFirstChild("Left Leg") or char:FindFirstChild("Right Leg")
        local legHeight = (leg and leg.Size.Y) or 2.0
        return legHeight + (root.Size.Y * 0.5)
    end
    local hip = (hum and hum.HipHeight and hum.HipHeight > 0.1) and hum.HipHeight or 2.0
    return hip + (root.Size.Y * 0.5)
end

local function getFloorY(referenceRoot, x, z)
    local refChar = referenceRoot.Parent
    local refHum = refChar and refChar:FindFirstChildOfClass("Humanoid")
    local refOffset = getRootOffsetFromFloor(refChar, refHum, referenceRoot)
    local fallbackFloor = referenceRoot.Position.Y - refOffset

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local excludeList = {}
    if illusionFolder then
        table.insert(excludeList, illusionFolder)
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then
            table.insert(excludeList, p.Character)
        end
    end
    params.FilterDescendantsInstances = excludeList

    local origin = Vector3.new(x, referenceRoot.Position.Y + 6, z)
    local result = Workspace:Raycast(origin, Vector3.new(0, -40, 0), params)
    if result then
        return result.Position.Y
    end
    return fallbackFloor
end

local function captureHealthyAppearance(player)
    local char = player and player.Character
    if not char then
        return nil
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    local head = char:FindFirstChild("Head")
    if not hum or hum.Health <= 0 or not root or not head then
        return cachedHealthyAppearances[player.UserId]
    end

    local prevArch = char.Archivable
    char.Archivable = true
    local template = char:Clone()
    char.Archivable = prevArch
    if not template then
        return cachedHealthyAppearances[player.UserId]
    end

    for _, desc in ipairs(template:GetDescendants()) do
        if desc:IsA("Script") or desc:IsA("LocalScript") or desc:IsA("Tool") then
            desc:Destroy()
        elseif desc:IsA("BasePart") then
            desc.LocalTransparencyModifier = 0
            desc.CanCollide = false
            desc.Anchored = false
            desc.AssemblyLinearVelocity = Vector3.zero
            desc.AssemblyAngularVelocity = Vector3.zero
        end
    end

    local oldCache = cachedHealthyAppearances[player.UserId]
    if oldCache then
        oldCache:Destroy()
    end
    cachedHealthyAppearances[player.UserId] = template
    return template
end

local function spawnFreshClone(sourcePlayer, cloneLabel, targetPos, facingYawCFrame)
    local template = captureHealthyAppearance(sourcePlayer)
    if not template then
        return nil
    end

    local clone = template:Clone()
    if not clone then
        return nil
    end

    clone.Name = cloneLabel or (sourcePlayer.Name .. "_Festival")

    local hum = clone:FindFirstChildOfClass("Humanoid")
    local root = clone:FindFirstChild("HumanoidRootPart") or clone.PrimaryPart
    if hum then
        hum.DisplayName = sourcePlayer.DisplayName
        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer
        hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOn
        hum.MaxHealth = 100
        hum.Health = 100
        hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
    end

    local folder = ensureIllusionFolder()
    clone.Parent = folder

    if root and targetPos then
        local offsetFromFloor = getRootOffsetFromFloor(clone, hum, root)
        local exactRootY = targetPos.Y + offsetFromFloor
        local finalCf = CFrame.new(targetPos.X, exactRootY, targetPos.Z) * (facingYawCFrame or CFrame.identity)
        clone:PivotTo(finalCf)
        root.Anchored = true
    end

    return clone
end

local function gatherTargetSources()
    local targets = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            if captureHealthyAppearance(player) then
                table.insert(targets, {
                    player = player,
                    label = player.DisplayName,
                })
            end
        end
    end

    if #targets == 0 and CONFIG.SoloFallbackClones > 0 and LocalPlayer.Character then
        captureHealthyAppearance(LocalPlayer)
        for i = 1, CONFIG.SoloFallbackClones do
            table.insert(targets, {
                player = LocalPlayer,
                label = LocalPlayer.DisplayName .. " #" .. i,
            })
        end
    end

    return targets
end

local function spawnHugeOceanInFront(anchorRoot, folder)
    local yawOnly = anchorRoot.CFrame - anchorRoot.CFrame.Position
    local floorY = getFloorY(anchorRoot, anchorRoot.Position.X, anchorRoot.Position.Z)
    local oceanCenter = (anchorRoot.CFrame * CFrame.new(0, 0, -(CONFIG.OceanLength * 0.5 + 3.2))).Position
    local surfaceY = floorY + 0.18

    deepBed.Anchored = true
    deepBed.CanCollide = false
    deepBed.CFrame = CFrame.new(oceanCenter.X, surfaceY - 1.1, oceanCenter.Z) * yawOnly

    waterSurface.Transparency = 0.24
    waterSurface.Reflectance = 0.45
    waterSurface.Anchored = true
    waterSurface.CanCollide = false
    waterSurface.CFrame = CFrame.new(oceanCenter.X, surfaceY, oceanCenter.Z) * yawOnly
    waterSurface.Parent = oceanModel

    local shorePos = (anchorRoot.CFrame * CFrame.new(0, 0, -3.4)).Position
    local foam = Instance.new("Part")
    foam.Name = "ShorelineFoam"
    foam.Size = Vector3.new(CONFIG.OceanWidth, 0.14, 1.8)
    foam.Color = Color3.fromRGB(224, 242, 254)
    foam.Material = Enum.Material.SmoothPlastic
    foam.Transparency = 0.45
    foam.Anchored = true
    foam.CanCollide = false
    foam.CFrame = CFrame.new(shorePos.X, surfaceY + 0.1, shorePos.Z) * yawOnly
    foam.Parent = oceanModel

    local waveParts = {}
    for w = 1, 5 do
        local distZ = -(10 + w * 26)
        local wPos = (anchorRoot.CFrame * CFrame.new(0, 0, distZ)).Position
        local wave = Instance.new("Part")
        wave.Name = "OceanWave_" .. w
        wave.Size = Vector3.new(CONFIG.OceanWidth * 0.92, 0.08, 3.2)

            baseCf = wave.CFrame,
            phase = w * 1.1,

    local tWave = 0
    waveConn = RunService.Heartbeat:Connect(function(dt)

            waveConn:Disconnect()
            return

        tWave = tWave + dt
        for _, item in ipairs(waveParts) do
            local shift = math.sin(tWave * 1.6 + item.phase) * 1.8
            item.part.CFrame = item.baseCf * CFrame.new(0, math.sin(tWave * 2 + item.phase) * 0.04, shift)
        end
    end)

    return oceanModel, surfaceY
end

local function releaseKrathongToOcean(krathongModel, surfaceY, yawOnly, folder)
    if not krathongModel or not krathongModel.Parent then
        return
    end
    local primary = krathongModel.PrimaryPart
    if not primary then
        return
    end

    local startPos = Vector3.new(primary.Position.X, surfaceY + 0.24, primary.Position.Z)
    local startCf = CFrame.new(startPos) * yawOnly
    krathongModel:PivotTo(startCf)

    local ripple = Instance.new("Part")
    ripple.Name = "WaterRipple"
    ripple.Shape = Enum.PartType.Cylinder
    ripple.Size = Vector3.new(0.05, 1.6, 1.6)
    ripple.Color = Color3.fromRGB(224, 242, 254)
    ripple.Material = Enum.Material.Neon
    ripple.Transparency = 0.45
    ripple.Anchored = true
    ripple.CanCollide = false
    ripple.CFrame = CFrame.new(startPos.X, surfaceY + 0.15, startPos.Z) * CFrame.Angles(0, 0, math.rad(90))
    ripple.Parent = folder

    TweenService:Create(
        ripple,
        TweenInfo.new(1.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {
            Size = Vector3.new(0.05, 5.4, 5.4),
            Transparency = 1,
        }
    ):Play()
    task.delay(1.25, function()
        if ripple and ripple.Parent then
            ripple:Destroy()
        end
    end)

    task.spawn(function()
        local elapsed = 0
        local driftSpeed = 3.2 + (math.random() * 1.2)
        local bobSeed = math.random() * 6.28
        while krathongModel and krathongModel.Parent and isEnabled do
            local dt = RunService.Heartbeat:Wait()
            elapsed = elapsed + dt
            local forwardDist = elapsed * driftSpeed
            local bobY = math.sin(elapsed * 2.5 + bobSeed) * 0.08
            local swayX = math.sin(elapsed * 1.2 + bobSeed) * 0.45
            krathongModel:PivotTo(startCf * CFrame.new(swayX, bobY, -forwardDist))
        end
    end)
end

local function playSlowKrathongAnimation(char, motors, krathongModel, surfaceY, folder, currentToken)
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        return
    end
    local yawOnly = root.CFrame - root.CFrame.Position

    local t1 = 0
    while t1 < 0.9 and isEnabled and sessionToken == currentToken do
        local dt = RunService.RenderStepped:Wait()
        t1 = math.min(0.9, t1 + dt)
        local alpha = math.sin((t1 / 0.9) * (math.pi * 0.5))
        applyCeremonyPose(motors, alpha, 0)
        positionKrathongInHands(krathongModel, char, alpha, 0)
    end

    if not isEnabled or sessionToken ~= currentToken then
        return
    end

    local t2 = 0
    while t2 < 1.5 and isEnabled and sessionToken == currentToken do
        local dt = RunService.RenderStepped:Wait()
        t2 = math.min(1.5, t2 + dt)
        local alpha = t2 / 1.5
        local smooth = alpha * alpha * (3 - 2 * alpha)
        applyCeremonyPose(motors, 1 - smooth * 0.25, smooth)
        positionKrathongInHands(krathongModel, char, 1 - smooth * 0.25, smooth)
    end

    if not isEnabled or sessionToken ~= currentToken then
        return
    end

    releaseKrathongToOcean(krathongModel, surfaceY, yawOnly, folder)

    local t3 = 0
    while t3 < 0.75 and isEnabled and sessionToken == currentToken do
        local dt = RunService.RenderStepped:Wait()
        t3 = math.min(0.75, t3 + dt)
        local remain = 1 - (t3 / 0.75)
        applyCeremonyPose(motors, remain * 0.45, remain)
    end
end

-- =========================================================
-- ลูปหลักงานลอยกระทง (ไม่มีการตาย ซิงค์ตามความยาวเพลงวนซ้ำ)
-- =========================================================
local function startFestivalLoop(currentToken)
    local folder = ensureIllusionFolder()
    setRealPlayersHidden(true)
    setLocalPlayerFrozen(true)
    startBackgroundMusic(currentToken)
    setCinematicCameraEnabled(true)

    task.spawn(function()
        local myChar = LocalPlayer.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        while (not myRoot) and isEnabled and sessionToken == currentToken do
            task.wait(0.2)
            myChar = LocalPlayer.Character
            myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        end
        if not isEnabled or sessionToken ~= currentToken or not myRoot then
            return
        end

        setLocalPlayerFrozen(true)
        local anchorCf = myRoot.CFrame
        local yawOnly = anchorCf - anchorCf.Position

        -- เสกทะเลขนาดใหญ่ด้านหน้าเราทันที
        local _, oceanSurfaceY = spawnHugeOceanInFront(myRoot, folder)

        -- สร้างแถวหน้ากระดานแถวเดียวกับเรา (เลือดเต็ม 100% ทุกคน)
        local targets = gatherTargetSources()
        local lineupEntries = {}
        for idx, entry in ipairs(targets) do
            local tier = math.ceil(idx / 2)
            local side = (idx % 2 == 1) and -1 or 1
            local xOffset = side * tier * CONFIG.LineupSpacing
            local slotWorld = (anchorCf * CFrame.new(xOffset, 0, 0)).Position
            local slotFloorY = getFloorY(myRoot, slotWorld.X, slotWorld.Z)
            local spawnPos = Vector3.new(slotWorld.X, slotFloorY, slotWorld.Z)

            local clone = spawnFreshClone(entry.player, entry.label, spawnPos, yawOnly)
            if clone then
                table.insert(lineupEntries, {
                    clone = clone,
                    motors = snapshotCharacterMotors(clone),
                    krathong = nil,
                    phaseOffset = idx * 0.35,
                })
            end
        end

        savedSelfMotors = snapshotCharacterMotors(myChar)

        -- ตัวปล่อยโคมลอยเป็นระยะตลอดงาน
        task.spawn(function()
            while isEnabled and sessionToken == currentToken do
                spawnSkyLantern(myRoot, folder)
                task.wait(1.6)
            end
        end)

        -- วนลูปงานเทศกาลลอยกระทงยาวซิงค์ตามความยาวจริงของเพลง
        while isEnabled and sessionToken == currentToken do
            -- เตรียมกระทงใหม่ใส่มือทุกคนสำหรับรอบนี้
            for idx, item in ipairs(lineupEntries) do
                item.krathong = createRealisticKrathongModel("Krathong_" .. idx)
                item.krathong.Parent = folder
            end

            if CONFIG.GiveSelfKrathong then
                activeSelfKrathongModel = createRealisticKrathongModel("Krathong_Self")
                activeSelfKrathongModel.Parent = folder
            else
                activeSelfKrathongModel = nil
            end

            local songLen = getSyncedSongDuration()
            local tAct1End = songLen * 0.35
            local tSelfRelease = tAct1End
            local tGroupRelease = tAct1End + CONFIG.GroupFollowDelaySec
            local tAct3Start = math.max(tGroupRelease + 3.2, songLen * 0.60)
            local tFinaleStart = songLen * 0.88

            local selfTriggered = false
            local groupTriggered = false
            local selfDone = false
            local groupDone = false
            local finaleSpawned = false
            local cycleElapsed = 0

            while isEnabled and sessionToken == currentToken do
                local dt = RunService.RenderStepped:Wait()
                cycleElapsed = cycleElapsed + dt
                songLen = getSyncedSongDuration()
                tAct1End = songLen * 0.35
                tSelfRelease = tAct1End
                tGroupRelease = tAct1End + CONFIG.GroupFollowDelaySec
                tAct3Start = math.max(tGroupRelease + 3.2, songLen * 0.60)
                tFinaleStart = songLen * 0.88

                -- ช่วงที่ 1: รำวงถือกระทงโยกตัวเข้าจังหวะเพลง
                if cycleElapsed < tSelfRelease then
                    setCameraStage("act1_ramwong_krathong", anchorCf, #lineupEntries)
                    if activeSelfKrathongModel then
                        applyRamwongHoldPose(savedSelfMotors, cycleElapsed, 0)
                        local selfSway = math.sin(cycleElapsed * 3.1 * CONFIG.DanceEnergy) * 0.14
                        positionKrathongInHands(activeSelfKrathongModel, myChar, 0.25, 0, selfSway)
                    end
                    for _, item in ipairs(lineupEntries) do
                        applyRamwongHoldPose(item.motors, cycleElapsed, item.phaseOffset)
                        local swayX = math.sin((cycleElapsed * 3.1 * CONFIG.DanceEnergy) + item.phaseOffset) * 0.14
                        positionKrathongInHands(item.krathong, item.clone, 0.25, 0, swayX)
                    end

                -- ช่วงที่ 2: ตัวเราก้มลอยกระทง -> อีก 2 วิทุกคนปล่อยตามพร้อมกัน
                elseif cycleElapsed < tAct3Start then
                    if not selfTriggered then
                        selfTriggered = true
                        setCameraStage("act2_self_release", anchorCf, #lineupEntries)
                        task.spawn(function()
                            if activeSelfKrathongModel then
                                local k = activeSelfKrathongModel
                                activeSelfKrathongModel = nil
                                playSlowKrathongAnimation(myChar, savedSelfMotors, k, oceanSurfaceY, folder, currentToken)
                            end
                            selfDone = true
                        end)
                    end

                    if cycleElapsed < tGroupRelease then
                        for _, item in ipairs(lineupEntries) do
                            applyRamwongHoldPose(item.motors, cycleElapsed, item.phaseOffset)
                            local swayX = math.sin((cycleElapsed * 3.1 * CONFIG.DanceEnergy) + item.phaseOffset) * 0.14
                            positionKrathongInHands(item.krathong, item.clone, 0.25, 0, swayX)
                        end
                    elseif not groupTriggered then
                        groupTriggered = true
                        setCameraStage("act2_group_release", anchorCf, #lineupEntries)
                        local remaining = #lineupEntries
                        if remaining == 0 then
                            groupDone = true
                        end
                        for _, item in ipairs(lineupEntries) do
                            local k = item.krathong
                            item.krathong = nil
                            task.spawn(function()
                                if k then
                                    playSlowKrathongAnimation(item.clone, item.motors, k, oceanSurfaceY, folder, currentToken)
                                end
                                remaining = remaining - 1
                                if remaining <= 0 then
                                    groupDone = true
                                end
                            end)
                        end
                    end

                    if selfDone then
                        applyFestivalDancePose(savedSelfMotors, cycleElapsed, 0)
                    end
                    if groupDone then
                        for _, item in ipairs(lineupEntries) do
                            applyFestivalDancePose(item.motors, cycleElapsed, item.phaseOffset)
                        end
                    end

                -- ช่วงที่ 3 & 4: เต้นรำวงโบกมือฉลองริมทะเล + ปล่อยกระทงรอบส่งท้ายก่อนวนเพลง
                else
                    if cycleElapsed < tFinaleStart then
                        setCameraStage("act3_celebration_dance", anchorCf, #lineupEntries)
                    else
                        setCameraStage("act4_finale_wave", anchorCf, #lineupEntries)
                        if not finaleSpawned then
                            finaleSpawned = true
                            for idx, item in ipairs(lineupEntries) do
                                local finaleK = createRealisticKrathongModel("FinaleKrathong_" .. idx)
                                finaleK.Parent = folder
                                local root = item.clone:FindFirstChild("HumanoidRootPart")
                                if root then
                                    local pos = (root.CFrame * CFrame.new(0, 0, -2.2)).Position
                                    finaleK:PivotTo(CFrame.new(pos.X, oceanSurfaceY + 0.24, pos.Z) * yawOnly)
                                    releaseKrathongToOcean(finaleK, oceanSurfaceY, yawOnly, folder)
                                end
                            end
                            if CONFIG.GiveSelfKrathong then
                                local selfFinale = createRealisticKrathongModel("FinaleKrathong_Self")
                                selfFinale.Parent = folder
                                local pos = (myRoot.CFrame * CFrame.new(0, 0, -2.2)).Position
                                selfFinale:PivotTo(CFrame.new(pos.X, oceanSurfaceY + 0.24, pos.Z) * yawOnly)
                                releaseKrathongToOcean(selfFinale, oceanSurfaceY, yawOnly, folder)
                            end
                        end
                    end

                    applyFestivalDancePose(savedSelfMotors, cycleElapsed, 0)
                    for _, item in ipairs(lineupEntries) do
                        applyFestivalDancePose(item.motors, cycleElapsed, item.phaseOffset)
                    end
                end

                -- จบ 1 รอบความยาวเพลงแล้ว วนซ้ำรอบใหม่ต่อเนื่อง
                if cycleElapsed >= songLen then
                    break
                end
            end

            -- เคลียร์กระทงเก่าที่ลอยไปไกลแล้วเพื่อลดแลคก่อนเริ่มรอบเพลงถัดไป
            for _, child in ipairs(folder:GetChildren()) do
                if child:IsA("Model") and (child.Name:find("Krathong_") or child.Name:find("FinaleKrathong_")) then
                    child:Destroy()
                end
            end
        end
    end)
end

local function updateButtonVisual()
    if isEnabled then
        toggleBtn.BackgroundColor3 = Color3.fromRGB(180, 83, 9)
        stroke.Color = Color3.fromRGB(251, 191, 36)
        toggleBtn.Text = "กระทง: เปิด"
    else
        toggleBtn.BackgroundColor3 = Color3.fromRGB(17, 24, 39)
        stroke.Color = Color3.fromRGB(71, 85, 105)
        toggleBtn.Text = "กระทง: ปิด"
    end
end

local dragging = false
local dragStart = nil
local startPos = nil
local movedDistance = 0

toggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = toggleBtn.Position
        movedDistance = 0
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        movedDistance = delta.Magnitude
        toggleBtn.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if dragging then
            dragging = false
            if movedDistance < 6 then
                isEnabled = not isEnabled
                sessionToken = sessionToken + 1
                updateButtonVisual()
                if isEnabled then
                    startFestivalLoop(sessionToken)
                else
                    clearAllIllusions()
                end
            end
        end
    end
end)

updateButtonVisual()
