local function resolveAudioAsset()
    if cachedAudioAsset then
        return cachedAudioAsset
    end

    local getAsset = (type(getcustomasset) == "function" and getcustomasset)
        or (type(getsynasset) == "function" and getsynasset)

    if not getAsset or type(writefile) ~= "function" then
        return nil
    end

    local okGet, audioBytes = pcall(function()
        return game:HttpGet(CONFIG.MusicUrl)
    end)
    if not okGet or type(audioBytes) ~= "string" or #audioBytes < 128 then
        return nil
    end

    local okWrite = pcall(function()
        writefile(MUSIC_FILE, audioBytes)
    end)
    if not okWrite then
        return nil
    end

    local okAsset, assetUri = pcall(getAsset, MUSIC_FILE)
    if okAsset and type(assetUri) == "string" and assetUri ~= "" then
        cachedAudioAsset = assetUri
        return cachedAudioAsset
    end

    return nil
end

    if CONFIG.EnableMusic then

    end
end)

local function stopBackgroundMusic()
    if activeSound then
        pcall(function()
            activeSound:Stop()
            activeSound:Destroy()
        end)
        activeSound = nil
    end
end

local function startBackgroundMusic(currentToken)
    stopBackgroundMusic()
    if not CONFIG.EnableMusic then
        return nil
    end

    local sound = Instance.new("Sound")
    sound.Name = "LoyKratongFestival_BGM"
    sound.Looped = true
    sound.Volume = CONFIG.MusicVolume
    sound.Parent = SoundService
    activeSound = sound

    task.spawn(function()
        local assetId = resolveAudioAsset()
        if not assetId or not isRunning or sessionToken ~= currentToken or activeSound ~= sound then
            return
        end
        sound.SoundId = assetId
        pcall(function()
            sound:Play()
        end)
    end)

    return sound
end

local function fetchRobloxUserAvatar(username)
    local trimmed = username:match("^%s*(.-)%s*$")
    if not trimmed or trimmed == "" then
        return nil, "กรุณาใส่ชื่อ"
    end

    local okId, userId = pcall(function()
        return Players:GetUserIdFromNameAsync(trimmed)
    end)
    if not okId or not userId then
        return nil, "ไม่พบชื่อนี้"
    end

    local okName, exactName = pcall(function()
        return Players:GetNameFromUserIdAsync(userId)
    end)

    local okDesc, desc = pcall(function()
        return Players:GetHumanoidDescriptionFromUserId(userId)
    end)
    if not okDesc or not desc then
        return nil, "ดึงชุดไม่สำเร็จ"
    end

    return {
        username = (okName and exactName) or trimmed,
        userId = userId,
        description = desc,
    }, nil
end

local function createNormalizedAvatarModel(entry, spawnPos, yawOnly, folder)
    local model = nil

    if entry.description then
        local descClone = entry.description:Clone()
        if CONFIG.UniformHeight then
            -- บังคับให้ทุกคนตัวเท่ากันและยืนตรงไม่มีตัวเตี้ยจมพื้น
            descClone.HeightScale = 1
            descClone.WidthScale = 1
            descClone.DepthScale = 1
            descClone.HeadScale = 1
            descClone.ProportionScale = 0
            descClone.BodyTypeScale = 0
        end

        local okCreate, created = pcall(function()
            return Players:CreateHumanoidModelFromDescription(descClone, Enum.HumanoidRigType.R15)
        end)
        if okCreate and created then
            model = created
        end
    end

    -- กรณีสำรองถ้าไม่สามารถเรียก CreateHumanoidModelFromDescription ได้
    if not model and entry.fallbackChar then
        local prevArch = entry.fallbackChar.Archivable
        entry.fallbackChar.Archivable = true
        model = entry.fallbackChar:Clone()
        entry.fallbackChar.Archivable = prevArch
    end

    if not model then
        return nil
    end

    model.Name = entry.username or "Guest"

    for _, desc in ipairs(model:GetDescendants()) do
        if desc:IsA("Script") or desc:IsA("LocalScript") or desc:IsA("Tool") then
            desc:Destroy()
        elseif desc:IsA("BasePart") then
            desc.CanCollide = false
            desc.Anchored = false
            desc.AssemblyLinearVelocity = Vector3.zero
            desc.AssemblyAngularVelocity = Vector3.zero
        end
    end

    local hum = model:FindFirstChildOfClass("Humanoid")
    local root = model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart
    if hum then
        hum.DisplayName = entry.username or model.Name
        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer
        hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.DisplayWhenDamaged
        hum.MaxHealth = 100
        hum.Health = 100
        hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
    end

    model.Parent = folder

    if root and spawnPos then
        local offsetFromFloor = 3.0
        if hum and hum.RigType == Enum.HumanoidRigType.R6 then
            local leg = model:FindFirstChild("Left Leg") or model:FindFirstChild("Right Leg")
            offsetFromFloor = ((leg and leg.Size.Y) or 2.0) + (root.Size.Y * 0.5)
        elseif hum then
            local hip = (hum.HipHeight and hum.HipHeight > 0.1) and hum.HipHeight or 2.0
            offsetFromFloor = hip + (root.Size.Y * 0.5)
        end
        local finalCf = CFrame.new(spawnPos.X, spawnPos.Y + offsetFromFloor, spawnPos.Z) * (yawOnly or CFrame.identity)
        model:PivotTo(finalCf)
        root.Anchored = true
    end

    return model
end

local function setLocalPlayerFrozen(freeze)
    if freezeConnection then
        freezeConnection:Disconnect()
        freezeConnection = nil
    end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if not freeze then
        if savedPlayerState then
            if hum and hum.Parent then
                hum.WalkSpeed = savedPlayerState.WalkSpeed
                hum.JumpPower = savedPlayerState.JumpPower
                hum.JumpHeight = savedPlayerState.JumpHeight
                hum.AutoRotate = savedPlayerState.AutoRotate
            end
            if root and root.Parent then
                root.Anchored = savedPlayerState.Anchored
            end
            savedPlayerState = nil
        elseif root and root.Parent then
            root.Anchored = false
        end
        return
    end

    if not CONFIG.FreezeLocalPlayer then
        return
    end

    if hum and root then
        if not savedPlayerState then
            savedPlayerState = {
                WalkSpeed = hum.WalkSpeed,
                JumpPower = hum.JumpPower,
                JumpHeight = hum.JumpHeight,
                AutoRotate = hum.AutoRotate,
                Anchored = root.Anchored,
            }
        end
        hum.WalkSpeed = 0
        hum.JumpPower = 0
        hum.JumpHeight = 0
        hum.AutoRotate = false
        root.Anchored = true
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end

    freezeConnection = RunService.Heartbeat:Connect(function()
        if not isRunning or not CONFIG.FreezeLocalPlayer then
            return
        end
        local c = LocalPlayer.Character
        local h = c and c:FindFirstChildOfClass("Humanoid")
        local r = c and c:FindFirstChild("HumanoidRootPart")
        if h and h.Health > 0 then
            h.WalkSpeed = 0
            h.JumpPower = 0
            h.JumpHeight = 0
            h.AutoRotate = false
        end
        if r then
            r.Anchored = true
            r.AssemblyLinearVelocity = Vector3.zero
        end
    end)
end

local function setCameraFocus(mode, anchorCf, spotlightModel, rowCount)
    if currentCamMode ~= mode or currentSpotlightTarget ~= spotlightModel then
        camClock = 0
    end
    currentCamMode = mode
    if anchorCf then
        currentCamAnchor = anchorCf
    end
    currentSpotlightTarget = spotlightModel
    if rowCount then
        currentRowCount = rowCount
    end
end

local function setCinematicCameraEnabled(enabled)
    if cameraConnection then
        cameraConnection:Disconnect()
        cameraConnection = nil
    end

    local cam = Workspace.CurrentCamera
    if not cam then
        return
    end

    if not enabled then
        if savedCameraState then
            cam.CameraType = savedCameraState.CameraType
            cam.FieldOfView = savedCameraState.FieldOfView
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            cam.CameraSubject = hum or savedCameraState.CameraSubject
            savedCameraState = nil
        else
            cam.CameraType = Enum.CameraType.Custom
        end
        return
    end

    if not CONFIG.EnableCinematicCamera then
        return
    end

    if not savedCameraState then
        savedCameraState = {
            CameraType = cam.CameraType,
            CameraSubject = cam.CameraSubject,
            FieldOfView = cam.FieldOfView,
        }
    end

    cam.CameraType = Enum.CameraType.Scriptable

    cameraConnection = RunService.RenderStepped:Connect(function(dt)
        if not isRunning or not CONFIG.EnableCinematicCamera then
            return
        end
        local activeCam = Workspace.CurrentCamera
        if not activeCam then
            return
        end
        if activeCam.CameraType ~= Enum.CameraType.Scriptable then
            activeCam.CameraType = Enum.CameraType.Scriptable
        end

        camClock = camClock + dt
        local baseCf = currentCamAnchor
        local halfSpan = math.max(1, math.ceil(currentRowCount / 2)) * CONFIG.LineupSpacing
        local targetCf = activeCam.CFrame
        local targetFov = 58

        if currentCamMode == "wide_row" then
            local panX = math.sin(camClock * 0.45) * math.min(8, halfSpan * 0.5)
            local distZ = math.max(18, 13 + halfSpan * 0.85)
            local camPos = (baseCf * CFrame.new(panX, 6.4, -distZ)).Position
            local lookAt = (baseCf * CFrame.new(0, 2.0, -1.5)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 58
        elseif currentCamMode == "krathong_release" then
            local distZ = math.max(20, 14 + halfSpan * 0.85)
            local camPos = (baseCf * CFrame.new(math.sin(camClock * 0.35) * 3.0, 7.6, -distZ)).Position
            local lookAt = (baseCf * CFrame.new(0, 1.3, -4.5)).Position
            targetCf = CFrame.lookAt(camPos, lookAt)
            targetFov = 62
        elseif currentCamMode == "spotlight_person" and currentSpotlightTarget and currentSpotlightTarget.Parent then
            local root = currentSpotlightTarget:FindFirstChild("HumanoidRootPart") or currentSpotlightTarget.PrimaryPart
            if root then
                local charYaw = root.CFrame - root.CFrame.Position
                local charCenter = CFrame.new(root.Position) * charYaw
                -- แพนกล้องเฉียงซ้าย-ขวาเบาๆ หน้าตัวละครคนนั้นเพื่อโชว์ท่าเต้นเต็มตัว
                local orbitX = math.sin(camClock * 1.1) * 2.2
                local camPos = (charCenter * CFrame.new(orbitX, 2.2, -9.2)).Position
                local lookAt = (charCenter * CFrame.new(0, 0.4, 0)).Position
                targetCf = CFrame.lookAt(camPos, lookAt)
                targetFov = 50
            end
        end

        local alpha = math.clamp(dt * 4.2, 0, 1)
        activeCam.CFrame = activeCam.CFrame:Lerp(targetCf, alpha)
        activeCam.FieldOfView = activeCam.FieldOfView + (targetFov - activeCam.FieldOfView) * alpha
    end)
end

local function spawnSkyLantern(anchorRoot, folder)
    if not CONFIG.EnableSkyLanterns then
        return
    end
    local yawOnly = anchorRoot.CFrame - anchorRoot.CFrame.Position
    local offsetX = math.random(-70, 70)
    local offsetZ = -math.random(25, 115)
    local startPos = (anchorRoot.CFrame * CFrame.new(offsetX, math.random(2, 6), offsetZ)).Position

    local lantern = Instance.new("Part")
    lantern.Name = "SkyLantern"
    lantern.Size = Vector3.new(1.6, 2.2, 1.6)
    lantern.Color = Color3.fromRGB(255, 185, 75)
    lantern.Material = Enum.Material.Neon
    lantern.Transparency = 0.15
    lantern.Anchored = true
    lantern.CanCollide = false
    lantern.CFrame = CFrame.new(startPos) * yawOnly
    lantern.Parent = folder

    local light = Instance.new("PointLight")
    light.Color = Color3.fromRGB(255, 190, 85)
    light.Range = 14
    light.Brightness = 1.6
    light.Parent = lantern

    task.spawn(function()
        local t = 0
        local speedY = 2.4 + math.random() * 1.6
        local swaySeed = math.random() * 6.28
        while lantern and lantern.Parent and isRunning and t < 16 do
            local dt = RunService.Heartbeat:Wait()
            t = t + dt
            local curPos = startPos + Vector3.new(
                math.sin(t * 0.9 + swaySeed) * 2.2,
                t * speedY,
                math.cos(t * 0.7 + swaySeed) * 1.8
            )
            lantern.CFrame = CFrame.new(curPos) * yawOnly * CFrame.Angles(0, t * 0.35, 0)
            if t > 13 then
                lantern.Transparency = math.min(1, 0.15 + ((t - 13) / 3) * 0.85)
            end
        end
        if lantern and lantern.Parent then
            lantern:Destroy()
        end
    end)
end

local function getFloorY(referenceRoot, x, z)
    local fallbackFloor = referenceRoot.Position.Y - 3.0
    local refChar = referenceRoot.Parent
    local refHum = refChar and refChar:FindFirstChildOfClass("Humanoid")
    if refHum then
        if refHum.RigType == Enum.HumanoidRigType.R6 then
            local leg = refChar:FindFirstChild("Left Leg") or refChar:FindFirstChild("Right Leg")
            fallbackFloor = referenceRoot.Position.Y - (((leg and leg.Size.Y) or 2.0) + referenceRoot.Size.Y * 0.5)
        else
            local hip = (refHum.HipHeight and refHum.HipHeight > 0.1) and refHum.HipHeight or 2.0
            fallbackFloor = referenceRoot.Position.Y - (hip + referenceRoot.Size.Y * 0.5)
        end
    end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local excludeList = {}
    if illusionFolder then
        table.insert(excludeList, illusionFolder)
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then
            table.insert(excludeList, p.Character)
        end
    end
    params.FilterDescendantsInstances = excludeList

    local origin = Vector3.new(x, referenceRoot.Position.Y + 6, z)
    local result = Workspace:Raycast(origin, Vector3.new(0, -40, 0), params)
    if result then
        return result.Position.Y
    end
    return fallbackFloor
end

local function spawnHugeOceanInFront(anchorRoot, folder)
    local yawOnly = anchorRoot.CFrame - anchorRoot.CFrame.Position

            item.part.CFrame = item.baseCf * CFrame.new(0, math.sin(tWave * 2 + item.phase) * 0.04, shift)

local function snapshotCharacterMotors(char)
    local motors = {}
    for _, desc in ipairs(char:GetDescendants()) do
        if desc:IsA("Motor6D") then
            motors[desc.Name] = {
                motor = desc,
                origC0 = desc.C0,
            }
        end
    end
    return motors
end

local function restoreCharacterMotors(motors)
    for _, info in pairs(motors) do
        if info.motor and info.motor.Parent then
            info.motor.C0 = info.origC0
        end
    end
end

local function setMotorAngle(motors, name, cfAngle)
    local entry = motors[name]
    if entry and entry.motor and entry.motor.Parent then
        entry.motor.Transform = CFrame.identity
        entry.motor.C0 = entry.origC0 * cfAngle
    end
end

local function applyCeremonyPose(motors, prayWeight, bendWeight)
    local armRaise = prayWeight * 0.72 + bendWeight * 0.58
    local armInward = prayWeight * 0.36 + bendWeight * 0.25
    local elbowBend = prayWeight * 0.65 - bendWeight * 0.2
    local headBow = prayWeight * 0.28 + bendWeight * 0.35
    local torsoBend = bendWeight * 0.55

    setMotorAngle(motors, "RightShoulder", CFrame.Angles(armRaise, 0, -armInward))
    setMotorAngle(motors, "LeftShoulder", CFrame.Angles(armRaise, 0, armInward))
    setMotorAngle(motors, "RightElbow", CFrame.Angles(math.max(0, elbowBend), 0, 0))
    setMotorAngle(motors, "LeftElbow", CFrame.Angles(math.max(0, elbowBend), 0, 0))
    setMotorAngle(motors, "Waist", CFrame.Angles(-torsoBend, 0, 0))
    setMotorAngle(motors, "Neck", CFrame.Angles(-headBow, 0, 0))
    setMotorAngle(motors, "RightHip", CFrame.Angles(bendWeight * 0.22, 0, 0))
    setMotorAngle(motors, "LeftHip", CFrame.Angles(bendWeight * 0.22, 0, 0))
    setMotorAngle(motors, "RightKnee", CFrame.Angles(-bendWeight * 0.35, 0, 0))
    setMotorAngle(motors, "LeftKnee", CFrame.Angles(-bendWeight * 0.35, 0, 0))

    -- R6 fallback
    setMotorAngle(motors, "Right Shoulder", CFrame.Angles(0, -armInward, armRaise))
    setMotorAngle(motors, "Left Shoulder", CFrame.Angles(0, armInward, -armRaise))
    setMotorAngle(motors, "RootJoint", CFrame.Angles(-torsoBend, 0, 0))
end

-- สไตล์การเต้นเต็มตัว 6 แบบ (แต่ละคนสุ่มและสลับหลายท่าในรอบตัวเอง!)
local function applyUniqueFullBodyDance(motors, danceStyle, timeSec, phaseOffset)
    local t = (timeSec * 4.2 * CONFIG.DanceEnergy) + (phaseOffset or 0)
    local s1 = math.sin(t)
    local c1 = math.cos(t)
    local s2 = math.sin(t * 2)
    local absBounce = math.abs(s1)

    if danceStyle == 1 then
        -- ท่าที่ 1: รำวงไทยเต็มวง (วาดแขนสลับสูงต่ำ บิดเอว ย่อเข่าจังหวะ)
        setMotorAngle(motors, "Root", CFrame.new(s1 * 0.25, -absBounce * 0.18, 0) * CFrame.Angles(0, s1 * 0.22, 0))
        setMotorAngle(motors, "Waist", CFrame.Angles(0.05, s1 * 0.35, c1 * 0.14))
        setMotorAngle(motors, "RightShoulder", CFrame.Angles(1.5 + s1 * 0.7, 0.3 * c1, 0.55 + c1 * 0.25))
        setMotorAngle(motors, "LeftShoulder", CFrame.Angles(1.5 - s1 * 0.7, -0.3 * c1, -0.55 + c1 * 0.25))
        setMotorAngle(motors, "RightElbow", CFrame.Angles(0.85 + s2 * 0.3, 0, 0))
        setMotorAngle(motors, "LeftElbow", CFrame.Angles(0.85 - s2 * 0.3, 0, 0))
        setMotorAngle(motors, "RightWrist", CFrame.Angles(0, s2 * 0.45, c1 * 0.35))
        setMotorAngle(motors, "LeftWrist", CFrame.Angles(0, -s2 * 0.45, -c1 * 0.35))
        setMotorAngle(motors, "RightHip", CFrame.Angles(absBounce * 0.45, 0, 0.1))
        setMotorAngle(motors, "LeftHip", CFrame.Angles(absBounce * 0.45, 0, -0.1))
        setMotorAngle(motors, "RightKnee", CFrame.Angles(-absBounce * 0.75, 0, 0))
        setMotorAngle(motors, "LeftKnee", CFrame.Angles(-absBounce * 0.75, 0, 0))
        setMotorAngle(motors, "Neck", CFrame.Angles(0.08, -s1 * 0.2, c1 * 0.15))

    elseif danceStyle == 2 then
        -- ท่าที่ 2: ปาร์ตี้ดิสโก้ชูมือขึ้นฟ้าสลับข้าง + ก้าวขาเท่ๆ
        setMotorAngle(motors, "Root", CFrame.new(0, math.abs(s2) * 0.22, 0) * CFrame.Angles(0, c1 * 0.25, s1 * 0.12))
        setMotorAngle(motors, "Waist", CFrame.Angles(-0.08, s1 * 0.42, s1 * 0.18))
        setMotorAngle(motors, "RightShoulder", CFrame.Angles(2.35 + s1 * 0.65, 0, 0.45))
        setMotorAngle(motors, "LeftShoulder", CFrame.Angles(0.65 - s1 * 0.85, 0, -0.45))
        setMotorAngle(motors, "RightElbow", CFrame.Angles(0.35 + absBounce * 0.4, 0, 0))
        setMotorAngle(motors, "LeftElbow", CFrame.Angles(0.95 - absBounce * 0.3, 0, 0))
        setMotorAngle(motors, "RightHip", CFrame.Angles(math.max(0, s1) * 0.65, 0, 0.12))
        setMotorAngle(motors, "LeftHip", CFrame.Angles(math.max(0, -s1) * 0.65, 0, -0.12))
        setMotorAngle(motors, "RightKnee", CFrame.Angles(-math.max(0, s1) * 0.9, 0, 0))
        setMotorAngle(motors, "LeftKnee", CFrame.Angles(-math.max(0, -s1) * 0.9, 0, 0))
        setMotorAngle(motors, "Neck", CFrame.Angles(s2 * 0.18, s1 * 0.25, 0))

    elseif danceStyle == 3 then
        -- ท่าที่ 3: ฮิปฮอปเบานซ์ เหวี่ยงแขนไขว้อก & ย่อตัวพริ้ว
        setMotorAngle(motors, "Root", CFrame.new(c1 * 0.35, -absBounce * 0.28, 0) * CFrame.Angles(-0.12, s1 * 0.3, 0))
        setMotorAngle(motors, "Waist", CFrame.Angles(-0.22 + s2 * 0.12, -s1 * 0.35, c1 * 0.15))
        setMotorAngle(motors, "RightShoulder", CFrame.Angles(1.15 + c1 * 0.55, 0, -0.55 + s1 * 0.45))
        setMotorAngle(motors, "LeftShoulder", CFrame.Angles(1.15 - c1 * 0.55, 0, 0.55 + s1 * 0.45))
        setMotorAngle(motors, "RightElbow", CFrame.Angles(1.1 + s2 * 0.35, 0, 0))
        setMotorAngle(motors, "LeftElbow", CFrame.Angles(1.1 - s2 * 0.35, 0, 0))
        setMotorAngle(motors, "RightHip", CFrame.Angles(0.45 + absBounce * 0.35, 0, 0.18))
        setMotorAngle(motors, "LeftHip", CFrame.Angles(0.45 + absBounce * 0.35, 0, -0.18))
        setMotorAngle(motors, "RightKnee", CFrame.Angles(-0.75 - absBounce * 0.45, 0, 0))
        setMotorAngle(motors, "LeftKnee", CFrame.Angles(-0.75 - absBounce * 0.45, 0, 0))
        setMotorAngle(motors, "Neck", CFrame.Angles(-0.15 + s2 * 0.15, c1 * 0.2, 0))

    elseif danceStyle == 4 then
        -- ท่าที่ 4: กระโดดตบมือเหนือหัวสุดร่าเริง
        local hop = math.max(0, math.sin(t * 1.35))
        setMotorAngle(motors, "Root", CFrame.new(0, hop * 0.95, 0) * CFrame.Angles(0, s1 * 0.15, 0))
        setMotorAngle(motors, "Waist", CFrame.Angles(hop * 0.12, 0, s1 * 0.12))
        local clapAngle = 0.25 + (1 - hop) * 0.55
        setMotorAngle(motors, "RightShoulder", CFrame.Angles(2.45, 0, -clapAngle))
        setMotorAngle(motors, "LeftShoulder", CFrame.Angles(2.45, 0, clapAngle))
        setMotorAngle(motors, "RightElbow", CFrame.Angles(0.55, 0, 0))
        setMotorAngle(motors, "LeftElbow", CFrame.Angles(0.55, 0, 0))
        setMotorAngle(motors, "RightHip", CFrame.Angles(hop * 0.4, 0, hop * 0.25))
        setMotorAngle(motors, "LeftHip", CFrame.Angles(hop * 0.4, 0, -hop * 0.25))
        setMotorAngle(motors, "RightKnee", CFrame.Angles(-hop * 0.7, 0, 0))
        setMotorAngle(motors, "LeftKnee", CFrame.Angles(-hop * 0.7, 0, 0))
        setMotorAngle(motors, "Neck", CFrame.Angles(hop * 0.18, 0, s1 * 0.15))

    elseif danceStyle == 5 then
        -- ท่าที่ 5: โยกสะโพกซ้าย-ขวา เตะขาสลับ + แกว่งแขนกว้าง
        setMotorAngle(motors, "Root", CFrame.new(s1 * 0.55, absBounce * 0.15, 0) * CFrame.Angles(0, 0, -s1 * 0.16))
        setMotorAngle(motors, "Waist", CFrame.Angles(0.06, -s1 * 0.45, s1 * 0.22))
        setMotorAngle(motors, "RightShoulder", CFrame.Angles(0.9 + c1 * 0.95, 0, 0.75 + s1 * 0.3))
        setMotorAngle(motors, "LeftShoulder", CFrame.Angles(0.9 - c1 * 0.95, 0, -0.75 + s1 * 0.3))
        setMotorAngle(motors, "RightElbow", CFrame.Angles(0.45 + absBounce * 0.45, 0, 0))
        setMotorAngle(motors, "LeftElbow", CFrame.Angles(0.45 + absBounce * 0.45, 0, 0))
        setMotorAngle(motors, "RightHip", CFrame.Angles(math.max(0, s1) * 0.85, 0, 0.22))
        setMotorAngle(motors, "LeftHip", CFrame.Angles(math.max(0, -s1) * 0.85, 0, -0.22))
        setMotorAngle(motors, "RightKnee", CFrame.Angles(-0.25, 0, 0))
        setMotorAngle(motors, "LeftKnee", CFrame.Angles(-0.25, 0, 0))
        setMotorAngle(motors, "Neck", CFrame.Angles(0, s1 * 0.3, -s1 * 0.16))

    else
        -- ท่าที่ 6: หมุนตัว 360 องศา + โบกมือสองข้างให้กล้อง
        local spinYaw = (timeSec * 2.8 * CONFIG.DanceEnergy) + (phaseOffset or 0)
        setMotorAngle(motors, "Root", CFrame.new(0, absBounce * 0.25, 0) * CFrame.Angles(0, spinYaw, 0))
        setMotorAngle(motors, "Waist", CFrame.Angles(-0.08, s1 * 0.25, c1 * 0.12))
        setMotorAngle(motors, "RightShoulder", CFrame.Angles(2.15, 0, 0.5 + s2 * 0.35))
        setMotorAngle(motors, "LeftShoulder", CFrame.Angles(2.15, 0, -0.5 - s2 * 0.35))
        setMotorAngle(motors, "RightElbow", CFrame.Angles(0.65 + s2 * 0.4, 0, 0))
        setMotorAngle(motors, "LeftElbow", CFrame.Angles(0.65 - s2 * 0.4, 0, 0))
        setMotorAngle(motors, "RightHip", CFrame.Angles(absBounce * 0.35, 0, 0.1))
        setMotorAngle(motors, "LeftHip", CFrame.Angles(absBounce * 0.35, 0, -0.1))
        setMotorAngle(motors, "RightKnee", CFrame.Angles(-absBounce * 0.55, 0, 0))
        setMotorAngle(motors, "LeftKnee", CFrame.Angles(-absBounce * 0.55, 0, 0))
        setMotorAngle(motors, "Neck", CFrame.Angles(0.1, s1 * 0.2, 0))
    end

    -- รองรับกรณีตัวละคร LocalPlayer เป็น R6
    setMotorAngle(motors, "Right Shoulder", CFrame.Angles(0, s1 * 0.4, 1.55 + c1 * 0.65))
    setMotorAngle(motors, "Left Shoulder", CFrame.Angles(0, -s1 * 0.4, -1.55 + c1 * 0.65))
    setMotorAngle(motors, "Right Hip", CFrame.Angles(0, 0, s1 * 0.45))
    setMotorAngle(motors, "Left Hip", CFrame.Angles(0, 0, s1 * 0.45))
    if danceStyle == 6 then
        setMotorAngle(motors, "RootJoint", CFrame.Angles(0, 0, (timeSec * 2.8 * CONFIG.DanceEnergy)))
    else
        setMotorAngle(motors, "RootJoint", CFrame.Angles(-0.06, s1 * 0.15, s1 * 0.25))
    end
end

local function positionKrathongInHands(krathongModel, char, prayWeight, bendWeight, swayOffset)
    if not krathongModel or not krathongModel.Parent or not char then
        return
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    local chest = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or root
    if not root or not chest then
        return
    end

    local forwardDist = 1.55 + (bendWeight * 1.35)
    local heightOffset = 0.08 + (prayWeight * 0.45) - (bendWeight * 1.45)
    local yawOnly = root.CFrame - root.CFrame.Position
    local anchorRef = CFrame.new(chest.Position) * yawOnly
    local targetPos = (anchorRef * CFrame.new(swayOffset or 0, heightOffset, -forwardDist)).Position
    krathongModel:PivotTo(CFrame.new(targetPos) * yawOnly)
end

local function releaseKrathongToOcean(krathongModel, surfaceY, yawOnly, folder)
    if not krathongModel or not krathongModel.Parent then
        return
    end
    local primary = krathongModel.PrimaryPart
    if not primary then
        return
    end

    local startPos = Vector3.new(primary.Position.X, surfaceY + 0.24, primary.Position.Z)
    local startCf = CFrame.new(startPos) * yawOnly
    krathongModel:PivotTo(startCf)

    local ripple = Instance.new("Part")
    ripple.Name = "WaterRipple"
    ripple.Shape = Enum.PartType.Cylinder
    ripple.Size = Vector3.new(0.05, 1.6, 1.6)
    ripple.Color = Color3.fromRGB(224, 242, 254)
    ripple.Material = Enum.Material.Neon
    ripple.Transparency = 0.45
    ripple.Anchored = true
    ripple.CanCollide = false
    ripple.CFrame = CFrame.new(startPos.X, surfaceY + 0.15, startPos.Z) * CFrame.Angles(0, 0, math.rad(90))
    ripple.Parent = folder

    TweenService:Create(
        ripple,
        TweenInfo.new(1.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
        {
            Size = Vector3.new(0.05, 5.4, 5.4),
            Transparency = 1,
        }
    ):Play()
    task.delay(1.25, function()
        if ripple and ripple.Parent then
            ripple:Destroy()
        end
    end)

    task.spawn(function()
        local elapsed = 0
        local driftSpeed = 3.2 + (math.random() * 1.2)
        local bobSeed = math.random() * 6.28
        while krathongModel and krathongModel.Parent and isRunning do
            local dt = RunService.Heartbeat:Wait()
            elapsed = elapsed + dt
            local forwardDist = elapsed * driftSpeed
            local bobY = math.sin(elapsed * 2.5 + bobSeed) * 0.08
            local swayX = math.sin(elapsed * 1.2 + bobSeed) * 0.45
            krathongModel:PivotTo(startCf * CFrame.new(swayX, bobY, -forwardDist))
        end
    end)
end

local function playSlowKrathongAnimation(char, motors, krathongModel, surfaceY, folder, currentToken)
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then
        return
    end
    local yawOnly = root.CFrame - root.CFrame.Position

    local t1 = 0
    while t1 < 0.85 and isRunning and sessionToken == currentToken do
        local dt = RunService.RenderStepped:Wait()
        t1 = math.min(0.85, t1 + dt)
        local alpha = math.sin((t1 / 0.85) * (math.pi * 0.5))
        applyCeremonyPose(motors, alpha, 0)
        positionKrathongInHands(krathongModel, char, alpha, 0)
    end

    if not isRunning or sessionToken ~= currentToken then
        return
    end

    local t2 = 0
    while t2 < 1.35 and isRunning and sessionToken == currentToken do
        local dt = RunService.RenderStepped:Wait()
        t2 = math.min(1.35, t2 + dt)
        local alpha = t2 / 1.35
        local smooth = alpha * alpha * (3 - 2 * alpha)
        applyCeremonyPose(motors, 1 - smooth * 0.25, smooth)
        positionKrathongInHands(krathongModel, char, 1 - smooth * 0.25, smooth)
    end

    if not isRunning or sessionToken ~= currentToken then
        return
    end

    releaseKrathongToOcean(krathongModel, surfaceY, yawOnly, folder)

    local t3 = 0
    while t3 < 0.65 and isRunning and sessionToken == currentToken do
        local dt = RunService.RenderStepped:Wait()
        t3 = math.min(0.65, t3 + dt)
        local remain = 1 - (t3 / 0.65)
        applyCeremonyPose(motors, remain * 0.45, remain)
    end
end

        if typed and typed ~= "" then
            if not row.pulledData then
                row.pullBtn.Text = "กำลังดึง..."
                local data, err = fetchRobloxUserAvatar(typed)
                if data then
                    row.pulledData = data
                    row.status = "pulled"
                    row.pullBtn.Text = "ดึงแล้ว"
                    row.pullBtn.BackgroundColor3 = Color3.fromRGB(21, 128, 61)
                else
                    row.pullBtn.Text = err or "ไม่พบชื่อ"
                    row.pullBtn.BackgroundColor3 = Color3.fromRGB(185, 28, 28)
                end
            end
            if row.pulledData then
                table.insert(roster, row.pulledData)
            end
        end
    end

    -- กรณีไม่ได้ใส่ชื่อใครเลย ให้ดึงตัวเราเองแสดงตัวอย่างเพื่อให้ทำงานได้ทันที
    if #roster == 0 then
        table.insert(roster, {
            username = LocalPlayer.DisplayName,
            userId = LocalPlayer.UserId,
            description = nil,
            fallbackChar = LocalPlayer.Character,
        })
    end

    return roster
end

-- ลูปหลัก: รอ 3 วิ -> ลอยกระทงทันที -> กล้องวนหาทีละคนเต้นสุ่ม -> วนซ้ำ
-- =========================================================
local function startFestivalShow(currentToken)
    local folder = ensureIllusionFolder()
    setLocalPlayerFrozen(true)
    startBackgroundMusic(currentToken)
    setCinematicCameraEnabled(true)

    task.spawn(function()
        local myChar = LocalPlayer.Character
        local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        while (not myRoot) and isRunning and sessionToken == currentToken do
            task.wait(0.2)
            myChar = LocalPlayer.Character
            myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
        end
        if not isRunning or sessionToken ~= currentToken or not myRoot then
            return
        end

        setLocalPlayerFrozen(true)
        local anchorCf = myRoot.CFrame
        local yawOnly = anchorCf - anchorCf.Position

        -- เสกทะเลใหญ่ด้านหน้าเรา
        local _, oceanSurfaceY = spawnHugeOceanInFront(myRoot, folder)

        -- ดึงอวาต้าทุกคนจากรายชื่อและสร้างแถวหน้ากระดานยืนตรงความสูงเท่ากัน
        local roster = collectRosterForRun()
        if not isRunning or sessionToken ~= currentToken then
            return
        end

        local dancers = {}
        for idx, entry in ipairs(roster) do
            local tier = math.ceil(idx / 2)
            local side = (idx % 2 == 1) and -1 or 1
            local xOffset = side * tier * CONFIG.LineupSpacing
            local slotWorld = (anchorCf * CFrame.new(xOffset, 0, 0)).Position
            local slotFloorY = getFloorY(myRoot, slotWorld.X, slotWorld.Z)
            local spawnPos = Vector3.new(slotWorld.X, slotFloorY, slotWorld.Z)

            local avatarModel = createNormalizedAvatarModel(entry, spawnPos, yawOnly, folder)
            if avatarModel then
                table.insert(dancers, {
                    model = avatarModel,
                    motors = snapshotCharacterMotors(avatarModel),
                    krathong = nil,
                    danceStyle = math.random(1, 6),
                    phaseOffset = idx * 0.65,
                    isSelf = false,
                })
            end
        end

        savedSelfMotors = snapshotCharacterMotors(myChar)

        -- ลำดับรายชื่อที่จะให้กล้องโฟกัสทีละคน (รวมตัวเราด้วย)
        local spotlightQueue = {}
        for _, d in ipairs(dancers) do
            table.insert(spotlightQueue, d)
        end
        table.insert(spotlightQueue, {
            model = myChar,
            motors = savedSelfMotors,
            danceStyle = math.random(1, 6),
            phaseOffset = 0,
            isSelf = true,
        })

        -- ปล่อยโคมลอยเป็นระยะตลอดงาน
        task.spawn(function()
            while isRunning and sessionToken == currentToken do
                spawnSkyLantern(myRoot, folder)
                task.wait(1.6)
            end
        end)

        -- วนซ้ำรอบ: [รอ 3 วิลอยกระทงทันที] -> [กล้องไปหาทีละคนเต้นสุ่มหลายท่า] -> [วนกลับลอยกระทง]
        while isRunning and sessionToken == currentToken do
            -- เตรียมกระทงตั้งตรงใส่มือทุกคน
            for idx, d in ipairs(dancers) do
                d.krathong = createRealisticKrathongModel("Krathong_" .. idx)
                d.krathong.Parent = folder
                applyCeremonyPose(d.motors, 0.2, 0)
                positionKrathongInHands(d.krathong, d.model, 0.2, 0, 0)
            end
            if CONFIG.GiveSelfKrathong then
                activeSelfKrathongModel = createRealisticKrathongModel("Krathong_Self")
                activeSelfKrathongModel.Parent = folder
                applyCeremonyPose(savedSelfMotors, 0.2, 0)
                positionKrathongInHands(activeSelfKrathongModel, myChar, 0.2, 0, 0)
            end

            -- 1) รอ 3 วินาที แล้วลอยกระทงทันที
            setCameraFocus("wide_row", anchorCf, nil, #dancers)
            local waitElapsed = 0
            while waitElapsed < CONFIG.PreFloatWaitSec and isRunning and sessionToken == currentToken do
                local dt = RunService.RenderStepped:Wait()
                waitElapsed = waitElapsed + dt
                local gentleSway = math.sin(waitElapsed * 2.4) * 0.08
                if activeSelfKrathongModel then
                    applyCeremonyPose(savedSelfMotors, 0.22 + gentleSway * 0.4, 0)
                    positionKrathongInHands(activeSelfKrathongModel, myChar, 0.22, 0, gentleSway)
                end
                for _, d in ipairs(dancers) do
                    applyCeremonyPose(d.motors, 0.22 + gentleSway * 0.4, 0)
                    positionKrathongInHands(d.krathong, d.model, 0.22, 0, gentleSway)
                end
            end

            if not isRunning or sessionToken ~= currentToken then
                return
            end

            -- 2) ทุกคนปล่อยกระทงลงทะเลทันทีพร้อมกัน
            setCameraFocus("krathong_release", anchorCf, nil, #dancers)
            local releaseDoneCount = 0
            local totalToRelease = #dancers + (activeSelfKrathongModel and 1 or 0)

            if activeSelfKrathongModel then
                local selfK = activeSelfKrathongModel
                activeSelfKrathongModel = nil
                task.spawn(function()
                    playSlowKrathongAnimation(myChar, savedSelfMotors, selfK, oceanSurfaceY, folder, currentToken)
                    releaseDoneCount = releaseDoneCount + 1
                end)
            end

            for _, d in ipairs(dancers) do
                local k = d.krathong
                d.krathong = nil
                task.spawn(function()
                    if k then
                        playSlowKrathongAnimation(d.model, d.motors, k, oceanSurfaceY, folder, currentToken)
                    end
                    releaseDoneCount = releaseDoneCount + 1
                end)
            end

            while releaseDoneCount < totalToRelease and isRunning and sessionToken == currentToken do
                RunService.Heartbeat:Wait()
            end

            if not isRunning or sessionToken ~= currentToken then
                return
            end

            -- 3) มุมกล้องไปหาแต่ละตัวละครทีละคน คนละ 3 วิ + สุ่มท่าเต้นเต็มตัวหลากหลายคนละหลายๆ ที
            for _, d in ipairs(dancers) do
                d.danceStyle = math.random(1, 6)
            end
            spotlightQueue[#spotlightQueue].danceStyle = math.random(1, 6)

            local globalDanceClock = 0
            for _, targetDancer in ipairs(spotlightQueue) do
                if not isRunning or sessionToken ~= currentToken then
                    return
                end

                -- สุ่มเปลี่ยนท่าเต้นใหม่ให้คนนี้เด่นๆ และสลับอีกท่ากลางช่วงเพื่อเต้นหลายๆ ท่า
                targetDancer.danceStyle = math.random(1, 6)
                local secondComboStyle = ((targetDancer.danceStyle + math.random(1, 4) - 1) % 6) + 1
                local switchedCombo = false

                setCameraFocus("spotlight_person", anchorCf, targetDancer.model, #dancers)

                local spotElapsed = 0
                while spotElapsed < CONFIG.SpotlightPerPersonSec and isRunning and sessionToken == currentToken do
                    local dt = RunService.RenderStepped:Wait()
                    spotElapsed = spotElapsed + dt
                    globalDanceClock = globalDanceClock + dt

                    if not switchedCombo and spotElapsed >= (CONFIG.SpotlightPerPersonSec * 0.5) then
                        switchedCombo = true
                        targetDancer.danceStyle = secondComboStyle
                    end

                    -- ตัวเราและทุกคนเต้นท่าแตกต่างกันแยกจากกันตลอดเวลา
                    local selfEntry = spotlightQueue[#spotlightQueue]
                    applyUniqueFullBodyDance(savedSelfMotors, selfEntry.danceStyle, globalDanceClock, 0)
                    for _, d in ipairs(dancers) do
                        applyUniqueFullBodyDance(d.motors, d.danceStyle, globalDanceClock, d.phaseOffset)
                    end
                end
            end

            -- ล้างกระทงเก่าที่ลอยไกลก่อนวนกลับรอบลอยกระทงใหม่
            for _, child in ipairs(folder:GetChildren()) do
                if child:IsA("Model") and child.Name:find("Krathong_") then
                    child:Destroy()
                end
            end
        end
    end)
end

local function setMenuVisibleAnimated(show)
    isMenuOpen = show
    if show then
        menuFrame.Visible = true
        menuScale.Scale = 0.82
        menuFrame.BackgroundTransparency = 0.35
        TweenService:Create(
            menuScale,
            TweenInfo.new(0.32, Enum.EasingStyle.Back, Enum.EasingDirection.Out),
            { Scale = 1 }
        ):Play()
        TweenService:Create(
            menuFrame,
            TweenInfo.new(0.24, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            { BackgroundTransparency = 0 }
        ):Play()
    else
        local shrinkTween = TweenService:Create(
            menuScale,
            TweenInfo.new(0.22, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { Scale = 0.82 }
        )
        TweenService:Create(
            menuFrame,
            TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In),
            { BackgroundTransparency = 0.45 }
        ):Play()
        shrinkTween:Play()
        shrinkTween.Completed:Connect(function()
            if not isMenuOpen then
                menuFrame.Visible = false
            end
        end)
    end
end

                pullBtn.Text = err or "ไม่พบชื่อ"
                pullBtn.BackgroundColor3 = Color3.fromRGB(185, 28, 28)
            end
            updatePulledCounter()
        end)
    end)

    addPlusBtn.MouseButton1Click:Connect(function()
        addInviteRow("")
    end)

    removeBtn.MouseButton1Click:Connect(function()
        if #inviteRows <= 1 then
            nameBox.Text = ""
            rowEntry.pulledData = nil
            rowEntry.status = "idle"
            pullBtn.Text = "ดึงตัว"
            pullBtn.BackgroundColor3 = Color3.fromRGB(22, 163, 74)
            updatePulledCounter()
            return
        end
        for i, item in ipairs(inviteRows) do
            if item.rowId == rowId then
                table.remove(inviteRows, i)
                break
            end
        end
        rowFrame:Destroy()
        updatePulledCounter()
    end)
end

    for _, uname in ipairs(CONFIG.InitialUsernames) do
        addInviteRow(uname)

-- ปุ่มรันทันที & ปุ่มหยุด
local function updateRunStatusVisual()
    if isRunning then
        runBtn.Text = "กำลังทำงาน..."
        runBtn.BackgroundColor3 = Color3.fromRGB(21, 128, 61)
        stopBtn.BackgroundColor3 = Color3.fromRGB(185, 28, 28)

        toggleStroke.Color = Color3.fromRGB(34, 197, 94)
    else
        runBtn.Text = "รันทันที"
        runBtn.BackgroundColor3 = Color3.fromRGB(217, 119, 6)
        stopBtn.BackgroundColor3 = Color3.fromRGB(51, 65, 85)
        menuToggleBtn.Text = "เมนูลอยกระทง"
        toggleStroke.Color = Color3.fromRGB(245, 158, 11)
    end
end

    if isRunning then
        sessionToken = sessionToken + 1
        clearFestivalScene()
    end

    sessionToken = sessionToken + 1
    updateRunStatusVisual()

    if CONFIG.AutoCollapseOnRun then
        setMenuVisibleAnimated(false)
    end

    startFestivalShow(sessionToken)
end)

stopBtn.MouseButton1Click:Connect(function()
    isRunning = false
    sessionToken = sessionToken + 1
    clearFestivalScene()
    updateRunStatusVisual()
end)

closeMenuBtn.MouseButton1Click:Connect(function()
    setMenuVisibleAnimated(false)
end)

-- ลากปุ่มเปิด-ปิดเมนูได้ทั้งเมาส์และทัชสกรีน
local dragging = false
local dragStart = nil
local startPos = nil
local movedDistance = 0

menuToggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = menuToggleBtn.Position
        movedDistance = 0
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        movedDistance = delta.Magnitude
        menuToggleBtn.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        if dragging then
            dragging = false
            if movedDistance < 6 then
                setMenuVisibleAnimated(not isMenuOpen)
            end
        end
    end
end)

-- เปิดเมนูด้วยอนิเมชั่นทันทีตอนรันสคริปต์
setMenuVisibleAnimated(true)
updateRunStatusVisual()
