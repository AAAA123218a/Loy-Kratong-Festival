-- Utility: Safe Part Builder
local function createPart(props)
	local p = Instance.new("Part")
	p.Anchored = true
	p.CanCollide = false
	p.CanQuery = false
	p.CanTouch = false
	p.CastShadow = false
	p.TopSurface = Enum.SurfaceType.Smooth
	p.BottomSurface = Enum.SurfaceType.Smooth
	p.Material = props.Material or Enum.Material.SmoothPlastic
	p.Shape = props.Shape or Enum.PartType.Block
	p.Size = props.Size or Vector3.new(1, 1, 1)
	p.Color = props.Color or Color3.new(1, 1, 1)
	p.Transparency = props.Transparency or 0
	if props.CFrame then
		p.CFrame = props.CFrame
	end
	if props.Name then
		p.Name = props.Name
	end
	p.Parent = props.Parent
	return p
end

-- Utility: Sound Setup (SoundService + Fallback Beat Envelope)
local festivalSound = nil
if CONFIG.SOUND_ID ~= "" then
	local ok, snd = pcall(function()
		local s = Instance.new("Sound")
		s.Name = "LoyKrathongFestivalAudio"
		s.SoundId = CONFIG.SOUND_ID
		s.Volume = CONFIG.SOUND_VOLUME
		s.Looped = true
		s.Parent = SoundService
		return s
	end)
	if ok and snd then
		festivalSound = snd
		task.spawn(function()
			pcall(function()
				if not festivalSound.IsLoaded then
					festivalSound.Loaded:Wait()
				end
				if state.running and not state.paused then
					festivalSound:Play()
				end
			end)
		end)
	end
end

local function getBeatLevel(t)
	local loudness = 0
	if festivalSound and festivalSound.IsPlaying then
		loudness = festivalSound.PlaybackLoudness
	end
	if loudness <= 1 then
		local pulse = math.abs(math.sin(t * 6.283))
		loudness = (pulse ^ 2) * 280
	end
	return math.clamp(loudness / 300, 0, 1.4)
end

		Color = Color3.fromRGB(118, 196, 88),
		Material = Enum.Material.SmoothPlastic,
		Parent = model,
	})

	local marigoldCenter = createPart({
		Name = "MarigoldFlower",
		Shape = Enum.PartType.Ball,
		Size = Vector3.new(0.95, 0.45, 0.95),
		Color = Color3.fromRGB(255, 184, 48),
		Material = Enum.Material.SmoothPlastic,
		Parent = model,
	})

	local candle = createPart({
		Name = "CandleStick",
		Shape = Enum.PartType.Cylinder,
		Size = Vector3.new(0.9, 0.24, 0.24),
		Color = Color3.fromRGB(255, 246, 210),
		Material = Enum.Material.SmoothPlastic,

	})

		Name = "CandleFlame",
		Shape = Enum.PartType.Ball,
		Size = Vector3.new(0.3, 0.44, 0.3),
		Color = Color3.fromRGB(255, 198, 64),
		Material = Enum.Material.Neon,

	light.Color = Color3.fromRGB(255, 196, 84)
	light.Range = 9
	light.Brightness = 1.4

	sparks.Name = "CandleSparks"

	sparks.Color = ColorSequence.new(Color3.fromRGB(255, 224, 110), Color3.fromRGB(255, 148, 64))
	sparks.Enabled = false

		model = model,

local function spawnRiverRipple(originPos, color, maxRadius)
	if not CONFIG.ENABLE_WORLD_3D then return end
	local ring = createPart({
		Name = "MoonRipple",
		Shape = Enum.PartType.Cylinder,
		Size = Vector3.new(0.16, 2.2, 2.2),
		Color = color,
		Material = Enum.Material.Neon,
		Transparency = 0.25,
		CFrame = CFrame.new(originPos) * CFrame.Angles(0, 0, math.pi * 0.5),
		Parent = worldFolder,
	})
	local tween = TweenService:Create(
		ring,
		TweenInfo.new(0.95, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{
			Size = Vector3.new(0.06, maxRadius, maxRadius),
			Transparency = 1,
		}
	)
	tween:Play()
	Debris:AddItem(ring, 1.0)
end

local function spawnFireworkOrb(originPos, velocity, color)
	if not CONFIG.ENABLE_WORLD_3D then return end
	local orb = createPart({
		Name = "FireworkOrb",
		Shape = Enum.PartType.Ball,
		Size = Vector3.new(0.65, 0.65, 0.65),
		Color = color,
		Material = Enum.Material.Neon,
		Transparency = 0.05,
		CFrame = CFrame.new(originPos),
		Parent = worldFolder,
	})
	task.spawn(function()
		local startTime = tick()
		local pos = originPos
		local vel = velocity
		while state.running and orb.Parent and (tick() - startTime) < 0.95 do
			local dt = RunService.RenderStepped:Wait()
			vel = vel + Vector3.new(0, -28 * dt, 0)
			pos = pos + vel * dt
			local age = (tick() - startTime) / 0.95
			orb.CFrame = CFrame.new(pos)
			orb.Transparency = 0.05 + age * 0.95
			orb.Size = Vector3.new(0.65 + age * 0.6, 0.65 + age * 0.6, 0.65 + age * 0.6)
		end
		if orb.Parent then orb:Destroy() end
	end)
end

local function updateKrathongPose(kr, cf, visible, scaleMul, emitSparks, beat)
	local s = scaleMul or 1
	if not visible then
		kr.base.Transparency = 1
		kr.leafCrown.Transparency = 1
		kr.marigoldCenter.Transparency = 1
		kr.candle.Transparency = 1
		kr.flame.Transparency = 1
		kr.light.Enabled = false
		kr.sparks.Enabled = false
		for _, p in ipairs(kr.petals) do p.part.Transparency = 1 end
		return
	end
	kr.base.Transparency = 0
	kr.leafCrown.Transparency = 0
	kr.marigoldCenter.Transparency = 0
	kr.candle.Transparency = 0
	kr.flame.Transparency = 0
	kr.light.Enabled = true
	kr.light.Brightness = 1.2 + (beat or 0) * 0.9
	kr.sparks.Enabled = emitSparks
	kr.sparks.Rate = 10 * CONFIG.INTENSITY_SCALE

	local cylRot = CFrame.Angles(0, 0, math.pi * 0.5)
	kr.base.Size = Vector3.new(0.48, 2.35, 2.35) * s
	kr.leafCrown.Size = Vector3.new(0.24, 2.55, 2.55) * s
	kr.marigoldCenter.Size = Vector3.new(0.95, 0.45, 0.95) * s
	kr.candle.Size = Vector3.new(0.9, 0.24, 0.24) * s
	local flicker = 1 + math.sin(tick() * 14) * 0.12 + (beat or 0) * 0.18
	kr.flame.Size = Vector3.new(0.3, 0.44 * flicker, 0.3) * s

	kr.base.CFrame = cf * cylRot
	kr.leafCrown.CFrame = cf * CFrame.new(0, 0.22 * s, 0) * cylRot
	kr.marigoldCenter.CFrame = cf * CFrame.new(0, 0.35 * s, 0)
	kr.candle.CFrame = cf * CFrame.new(0, 0.72 * s, 0) * cylRot
	kr.flame.CFrame = cf * CFrame.new(0, 1.24 * s, 0)

	for _, p in ipairs(kr.petals) do
		p.part.Transparency = 0.05
		p.part.Size = Vector3.new(0.42, 0.26, 0.22) * s
		local px = math.cos(p.angle) * (1.05 * s)
		local pz = math.sin(p.angle) * (1.05 * s)
		p.part.CFrame = cf * CFrame.new(px, 0.42 * s, pz) * CFrame.Angles(0, -p.angle, 0.35)
	end
end

local function updateKhomLoiPose(ln, cf, visible, bodyColor)
	if not visible then
		ln.body.Transparency = 1
		ln.rim.Transparency = 1
		ln.core.Transparency = 1
		ln.embers.Enabled = false
		return
	end
	ln.body.Transparency = 0.12
	ln.rim.Transparency = 0
	ln.core.Transparency = 0
	ln.body.Color = bodyColor
	ln.embers.Enabled = true

	local cylRot = CFrame.Angles(0, 0, math.pi * 0.5)
	ln.body.CFrame = cf * cylRot
	ln.rim.CFrame = cf * CFrame.new(0, -1.05, 0) * cylRot
	ln.core.CFrame = cf * CFrame.new(0, -0.68, 0)
end

local function resetJointTransforms()
	for _, m in pairs(joints) do
		if m and m.Parent then
			m.Transform = CFrame.new()
		end
	end
end

local function bindCharacter(char)
	resetJointTransforms()
	joints = {}
	activeChar = char
	activeHumanoid = char:WaitForChild("Humanoid", 5)
	activeRoot = char:WaitForChild("HumanoidRootPart", 5)
	if not activeHumanoid or not activeRoot then return end

	isR15 = (activeHumanoid.RigType == Enum.HumanoidRigType.R15)
	for _, desc in ipairs(char:GetDescendants()) do
		if desc:IsA("Motor6D") then
			joints[desc.Name] = desc
		end
	end

	local anim = char:FindFirstChild("Animate")
	if anim and anim:IsA("LocalScript") and CONFIG.ENABLE_CHARACTER_ANIM then
		activeAnimateScript = anim
		if not anim.Disabled then
			anim.Disabled = true
			wasAnimateDisabled = true
		end
	end
end

if LocalPlayer.Character then
	task.spawn(bindCharacter, LocalPlayer.Character)
end
table.insert(connections, LocalPlayer.CharacterAdded:Connect(function(newChar)
	task.wait(0.25)
	if state.running then
		bindCharacter(newChar)
	end
end))

local function setJoint(name, cf)
	local m = joints[name]
	if m and m.Parent then
		m.Transform = cf
	end
end

local function applyCharacterPose(pose)
	if not CONFIG.ENABLE_CHARACTER_ANIM or not activeChar or not activeRoot then
		return
	end
	if isR15 then
		setJoint("Root", CFrame.new(0, pose.hover or 0, 0) * CFrame.Angles(0, pose.rootSpin or 0, 0))
		setJoint("Waist", CFrame.Angles(pose.torsoPitch or 0, pose.torsoYaw or 0, pose.torsoRoll or 0))
		setJoint("Neck", CFrame.Angles(pose.headPitch or 0, pose.headYaw or 0, 0))
		setJoint("LeftShoulder", CFrame.Angles(pose.lArmPitch or 0, pose.lArmYaw or 0, pose.lArmRoll or 0))
		setJoint("LeftElbow", CFrame.Angles(pose.lElbow or 0, 0, 0))
		setJoint("RightShoulder", CFrame.Angles(pose.rArmPitch or 0, pose.rArmYaw or 0, pose.rArmRoll or 0))
		setJoint("RightElbow", CFrame.Angles(pose.rElbow or 0, 0, 0))
		setJoint("LeftHip", CFrame.Angles(pose.lLegPitch or 0, 0, pose.lLegRoll or 0))
		setJoint("LeftKnee", CFrame.Angles(pose.lKnee or 0, 0, 0))
		setJoint("RightHip", CFrame.Angles(pose.rLegPitch or 0, 0, pose.rLegRoll or 0))
		setJoint("RightKnee", CFrame.Angles(pose.rKnee or 0, 0, 0))
	else
		setJoint("RootJoint", CFrame.new(0, 0, pose.hover or 0) * CFrame.Angles(0, 0, pose.rootSpin or 0))
		setJoint("Neck", CFrame.Angles(pose.headPitch or 0, 0, pose.headYaw or 0))
		setJoint("Left Shoulder", CFrame.Angles(0, pose.lArmYaw or 0, -(pose.lArmPitch or 0)) * CFrame.Angles(pose.lArmRoll or 0, 0, 0))
		setJoint("Right Shoulder", CFrame.Angles(0, pose.rArmYaw or 0, pose.rArmPitch or 0) * CFrame.Angles(pose.rArmRoll or 0, 0, 0))
		setJoint("Left Hip", CFrame.Angles(0, 0, -(pose.lLegPitch or 0)))
		setJoint("Right Hip", CFrame.Angles(0, 0, pose.rLegPitch or 0))
	end
end

-- ============================================================================
-- CLEANUP HANDLER
-- ============================================================================
local function cleanup()
	if not state.running then return end
	state.running = false

	for _, conn in ipairs(connections) do
		pcall(function() conn:Disconnect() end)
	end
	connections = {}

	resetJointTransforms()
	if wasAnimateDisabled and activeAnimateScript and activeAnimateScript.Parent then
		pcall(function() activeAnimateScript.Disabled = false end)
	end

	local cam = Workspace.CurrentCamera
	if cam then
		cam.CameraType = Enum.CameraType.Custom
	end

	if festivalSound then
		pcall(function()
			festivalSound:Stop()
			festivalSound:Destroy()
		end)
	end
	if colorCorrection and colorCorrection.Parent then
		colorCorrection:Destroy()
	end
	if worldFolder and worldFolder.Parent then
		worldFolder:Destroy()
	end
	if gui and gui.Parent then
		gui:Destroy()
	end
end

	if CONFIG.ENABLE_WORLD_3D then

				local r = 3.6 + (i % 5) * 1.15
				local yCycle = ((t * 2.2 + i * 0.35) % 9) - 2.5
				local pPos = centerPos + Vector3.new(math.cos(a) * r, yCycle, math.sin(a) * r)
				petal.CFrame = CFrame.new(pPos) * CFrame.Angles(t * 2 + i, a, t * 1.5)
				if i % 3 ~= 0 then
					petal.Color = palette.petal
				end
			end
		end
	end

	-- Cinematic 3D Orbit Camera
	if state.cameraEnabled then
		local cam = Workspace.CurrentCamera
		if cam then
			cam.CameraType = Enum.CameraType.Scriptable
			local orbitSpeed = (state.actIndex == 1) and 0.35 or ((state.actIndex == 5) and 0.9 or 0.55)
			local camAngle = t * orbitSpeed * dirSign + (state.actIndex * 0.8)
			local dist = (state.actIndex == 5) and 17 or 13.5
			local camHeight = (state.actIndex == 5) and 6.2 or (3.6 + math.sin(t * 0.9) * 1.4)
			local camPos = centerPos + Vector3.new(math.cos(camAngle) * dist, camHeight, math.sin(camAngle) * dist)
			cam.CFrame = CFrame.lookAt(camPos, centerPos + Vector3.new(0, 1.2, 0))
		end
	end
end))
